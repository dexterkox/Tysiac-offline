/* ============================================================================
   Tysiąc — łączenie bez sieci (Multipeer)
   ============================================================================
   Bonjour i WebSocket wymagają wspólnego Wi-Fi. Ten moduł nie wymaga niczego:
   telefony gadają bezpośrednio przez Bluetooth i Wi-Fi peer-to-peer, tak jak
   przy AirDropie. To jest tryb do samolotu.

   Interfejs jest celowo taki sam jak w `native-net.js`, żeby reszta gry nie
   musiała wiedzieć, którym kanałem lecą dane.
   ========================================================================== */

(function (global) {
  'use strict';

  function plugin() {
    const cap = global.Capacitor;
    return (cap && cap.Plugins && cap.Plugins.Multipeer) ? cap.Plugins.Multipeer : null;
  }

  const Nearby = {
    _peers: [],
    _listenersBound: false,
    _onPeersChange: null,
    _onData: null,
    _onConnected: null,
    _onDisconnected: null,

    isAvailable() {
      const cap = global.Capacitor;
      const native = !!(cap && typeof cap.isNativePlatform === 'function' && cap.isNativePlatform());
      return native && !!plugin();
    },

    _bind() {
      if (this._listenersBound) return;
      const p = plugin();
      if (!p) return;

      p.addListener('peerFound', (ev) => {
        if (!this._peers.some((x) => x.name === ev.name)) this._peers.push({ name: ev.name });
        if (this._onPeersChange) this._onPeersChange(this._peers.slice());
      });
      p.addListener('peerLost', (ev) => {
        this._peers = this._peers.filter((x) => x.name !== ev.name);
        if (this._onPeersChange) this._onPeersChange(this._peers.slice());
      });
      p.addListener('connected', (ev) => {
        if (this._onConnected) this._onConnected(ev.name);
      });
      p.addListener('disconnected', (ev) => {
        if (this._onDisconnected) this._onDisconnected(ev.name);
      });
      p.addListener('data', (ev) => {
        let obj = null;
        try { obj = JSON.parse(ev.data); } catch (e) { return; }
        if (this._onData) this._onData(ev.name, obj);
      });

      this._listenersBound = true;
    },

    /* ---------------- Gospodarz ---------------- */

    async host(displayName, handlers) {
      const p = plugin();
      if (!p) throw new Error('Tryb bez sieci dostępny tylko w aplikacji');
      this._bind();
      this._onData = handlers && handlers.onMessage;
      this._onConnected = handlers && handlers.onConnected;
      this._onDisconnected = handlers && handlers.onDisconnected;
      this._peers = [];
      return await p.startHosting({ displayName: 'Gra: ' + displayName });
    },

    /* ---------------- Wyszukiwanie ---------------- */

    async browse(displayName, onPeersChange, handlers) {
      const p = plugin();
      if (!p) throw new Error('Tryb bez sieci dostępny tylko w aplikacji');
      this._bind();
      this._onPeersChange = onPeersChange;
      this._onData = handlers && handlers.onMessage;
      this._onConnected = handlers && handlers.onConnected;
      this._onDisconnected = handlers && handlers.onDisconnected;
      this._peers = [];
      return await p.startBrowsing({ displayName: displayName });
    },

    async join(peerName) {
      const p = plugin();
      if (!p) throw new Error('Tryb bez sieci dostępny tylko w aplikacji');
      return await p.invite({ peer: peerName });
    },

    /* ---------------- Przesyłanie ---------------- */

    send(obj) {
      const p = plugin();
      if (!p) return;
      p.send({ data: JSON.stringify(obj) }).catch(() => {});
    },

    async stop() {
      const p = plugin();
      if (!p) return;
      this._peers = [];
      try { await p.stop(); } catch (e) {}
    }
  };

  global.Nearby = Nearby;
})(window);

</script>
<script>
/* ============================================================================
   Tysiąc — warstwa sieci lokalnej dla aplikacji natywnej
   ============================================================================
   W przeglądarce nie da się ani rozgłosić usługi, ani nasłuchiwać połączeń, więc
   parowanie wymagało kodów QR. W aplikacji natywnej oba te ograniczenia znikają:

     • gospodarz uruchamia serwer WebSocket na telefonie i rozgłasza go przez
       Bonjour (_tysiac._tcp),
     • pozostali przeglądają sieć i widzą listę gier — jak przeglądarka serwerów
       w CS — po czym dołączają jednym dotknięciem.

   Nic z tego nie wychodzi poza sieć lokalną, więc działa w samolocie.

   Ten plik celowo nie zakłada, że wtyczki są dostępne. Gdy ich nie ma (np. przy
   otwarciu gry w zwykłej przeglądarce), `NativeNet.isAvailable()` zwraca false,
   a gra korzysta ze starej ścieżki z kodami QR.
   ========================================================================== */

(function (global) {
  'use strict';

  const SERVICE_TYPE = '_tysiac._tcp.';
  const DOMAIN = 'local.';
  const PORT = 8777;

  function plugins() {
    const cap = global.Capacitor;
    return (cap && cap.Plugins) ? cap.Plugins : {};
  }
  function zeroconf() { return plugins().ZeroConf || global.ZeroConf || null; }
  function wsserver() { return global.cordova && global.cordova.plugins ? global.cordova.plugins.wsserver : null; }

  function isNativeShell() {
    const cap = global.Capacitor;
    return !!(cap && typeof cap.isNativePlatform === 'function' && cap.isNativePlatform());
  }

  const NativeNet = {
    /* True only inside the packaged app with both plugins present. */
    isAvailable() {
      return isNativeShell() && !!zeroconf() && !!wsserver();
    },

    /* ---------------- Gospodarz ---------------- */

    /* Starts the local server and announces the game on the network.
       onMessage(clientId, obj), onOpen(clientId), onClose(clientId) */
    async host(gameName, handlers) {
      if (!this.isAvailable()) throw new Error('Sieć lokalna dostępna tylko w aplikacji');
      const ws = wsserver();
      const zc = zeroconf();
      this._clients = {};

      await new Promise((resolve, reject) => {
        ws.start(PORT, {
          onFailure: (addr, port, reason) => reject(new Error(reason || 'Nie udało się uruchomić serwera')),
          onOpen: (conn) => {
            this._clients[conn.uuid] = conn;
            if (handlers && handlers.onOpen) handlers.onOpen(conn.uuid);
          },
          onMessage: (conn, msg) => {
            let obj = null;
            try { obj = JSON.parse(msg); } catch (e) { return; }
            if (handlers && handlers.onMessage) handlers.onMessage(conn.uuid, obj);
          },
          onClose: (conn) => {
            delete this._clients[conn.uuid];
            if (handlers && handlers.onClose) handlers.onClose(conn.uuid);
          },
          /* Called once the listening socket is up. */
          onStart: () => resolve()
        });
        /* Some builds of the plugin do not call onStart; give it a moment either way. */
        setTimeout(resolve, 1200);
      });

      await new Promise((resolve) => {
        zc.register(SERVICE_TYPE, DOMAIN, gameName, PORT, { gra: 'tysiac' }, () => resolve(), () => resolve());
      });

      this._hosting = true;
      return { port: PORT, name: gameName };
    },

    send(clientId, obj) {
      const ws = wsserver();
      if (!ws) return;
      try { ws.send({ uuid: clientId }, JSON.stringify(obj)); } catch (e) {}
    },

    broadcast(obj) {
      Object.keys(this._clients || {}).forEach((id) => this.send(id, obj));
    },

    async stopHosting() {
      const ws = wsserver(), zc = zeroconf();
      try { if (zc) zc.unregister(SERVICE_TYPE, DOMAIN, this._name || ''); } catch (e) {}
      try { if (ws) ws.stop(); } catch (e) {}
      this._hosting = false;
      this._clients = {};
    },

    /* ---------------- Przeglądanie sieci ---------------- */

    /* Continuously reports games found on the local network.
       onChange(list) where list = [{name, host, port, addresses}] */
    startBrowsing(onChange) {
      if (!this.isAvailable()) throw new Error('Sieć lokalna dostępna tylko w aplikacji');
      const zc = zeroconf();
      const found = {};
      this._browsing = true;

      zc.watch(SERVICE_TYPE, DOMAIN, (result) => {
        const action = result && result.action;
        const svc = result && result.service;
        if (!svc) return;
        const key = svc.name;
        if (action === 'resolved' || action === 'added') {
          const addr = (svc.ipv4Addresses && svc.ipv4Addresses[0]) || (svc.ipv6Addresses && svc.ipv6Addresses[0]);
          if (addr) {
            found[key] = { name: svc.name, host: addr, port: svc.port || PORT };
          }
        } else if (action === 'removed') {
          delete found[key];
        }
        onChange(Object.values(found));
      }, () => { /* watch errors are non-fatal; the list simply stays empty */ });
    },

    stopBrowsing() {
      const zc = zeroconf();
      try { if (zc) zc.unwatch(SERVICE_TYPE, DOMAIN); } catch (e) {}
      this._browsing = false;
    },

    /* ---------------- Dołączanie ---------------- */

    /* Connects to a game found while browsing. */
    join(service, handlers) {
      return new Promise((resolve, reject) => {
        const url = 'ws://' + service.host + ':' + (service.port || PORT);
        let socket;
        try { socket = new WebSocket(url); } catch (e) { reject(e); return; }

        const timer = setTimeout(() => {
          try { socket.close(); } catch (e) {}
          reject(new Error('Brak odpowiedzi od gospodarza'));
        }, 8000);

        socket.onopen = () => {
          clearTimeout(timer);
          this._socket = socket;
          if (handlers && handlers.onOpen) handlers.onOpen();
          resolve(socket);
        };
        socket.onmessage = (ev) => {
          let obj = null;
          try { obj = JSON.parse(ev.data); } catch (e) { return; }
          if (handlers && handlers.onMessage) handlers.onMessage(obj);
        };
        socket.onclose = () => { if (handlers && handlers.onClose) handlers.onClose(); };
        socket.onerror = () => { clearTimeout(timer); reject(new Error('Nie udało się połączyć')); };
      });
    },

    sendToHost(obj) {
      if (!this._socket || this._socket.readyState !== 1) return;
      try { this._socket.send(JSON.stringify(obj)); } catch (e) {}
    },

    leave() {
      try { if (this._socket) this._socket.close(); } catch (e) {}
      this._socket = null;
    }
  };

  global.NativeNet = NativeNet;
})(window);

  const order=['S','C','H','D'];
  hand.sort((a,b)=>{

    return rankIdx(b.rank)-rankIdx(a.rank);
  });
  return hand;
}

/* The host broadcasts a new state before it starts listening for the reply, so a quick tap on the

  if(!NET.actionQueue[rid]) NET.actionQueue[rid]=[];
  NET.actionQueue[rid].push(action);

    const isActive = (act!==null && act!==undefined && seatRealId(act)===p);
    const div=document.createElement('div');
    div.className='plaque'+(sitOut?' sitout':'')+(isActive?' active':'');
    const dealerMark = p===G.dealer ? ' 🂠' : '';
    let bombMark='';
    if(G.bombsEnabled){
      const left=BOMB_LIMIT-((G.bombsUsed&&G.bombsUsed[p])||0);
      bombMark='<div class="bombs">'+(left>0 ? '💣'.repeat(left) : '<span class="nobombs">brak bomb</span>')+'</div>';
    }
    div.innerHTML=`<div class="name">${G.playerNames[p]}${dealerMark}${sitOut?' (pauza)':''}</div><div class="score">${G.scores[p]}</div>${bombMark}`;
    sb.appendChild(div);
  }
  const btn=document.createElement('button');
  btn.className='menubtn'; btn.textContent='?';
  btn.title='wersja '+BUILD;
  btn.onclick=()=>toggleRules(true);
  sb.appendChild(btn);
}

/* One place decides whose move it is, so the marker is consistent across bidding, the musik
   choice, the exchange and the play itself. */
function activeSeatNow(){
  if(!G) return null;
  if(G.phase==='bidding' && G.bidding) return G.bidding.turn;
  if(G.phase==='musikpick' || G.phase==='exchange' || G.phase==='declare') return G.bidder;
  if(G.phase==='play') return currentTurnSeat();
  return null;
}

function renderHand(){
  const hz=el('hand');
  hz.innerHTML='';
  if(G.showGate) return;
  const hand=G.hands[G.focusSeat]||[];
  let legal=null, clickMode=null;
  if(G.phase==='play' && currentTurnSeat()===G.focusSeat){
    legal=legalMoves(G.focusSeat); clickMode='play';
  } else if(G.phase==='exchange' && G.exchange && G.exchange.step==='select' && G.focusSeat===G.bidder){
    clickMode='exchange';
  }

  const meldSuitsInHand = meldOptionsForHand(hand);
  /* Twelve cards (ten plus the musik) overflowed the row and the last ones became hard or
     impossible to tap, which looked like the game refusing a second selection. Tighten the
     overlap so the whole hand always fits on one line. */
  const n=hand.length;
  const avail=Math.min(window.innerWidth||390, 430)-16;
  const cardW=62;
  let overlap=20;
  if(n>1){
    const needed=(avail-cardW)/(n-1);
    overlap=Math.max(0, Math.min(44, cardW-needed)); /* allow a tighter fan for 12-card hands */
  }
  hz.style.flexWrap='nowrap';
  hand.forEach((card,i)=>{
    const wrap=cardFace(card);
    if(i>0) wrap.style.marginLeft=(-overlap)+'px';
    else wrap.style.marginLeft='0';
    /* Explicit stacking: without it the fan could paint out of order and cards looked as if they
       were showing through one another. */
    wrap.style.zIndex=String(10+i);
    if(clickMode==='play'){
      const isLegal=legal.some(c=>cid(c)===cid(card));
      wrap.className='card '+(isLegal?'playable':'disabled');
      if(isLegal && !cardSendLock) wrap.onclick=()=>{
        cardSendLock=true;
        setTimeout(()=>{ cardSendLock=false; }, 1200);
        respond(card);
      };
      if(cardSendLock) wrap.className='card disabled';
    } else if(clickMode==='exchange'){
      const sel=(G.exchange.selected||[]).find(s=>cid(s)===cid(card));
      wrap.className='card '+(sel?'picked':'playable');
      wrap.onclick=()=>toggleExchangeCard(card);
    } else {
      wrap.className='card';
    }
    /* Mark the card handed over by the bidder so it is obvious what changed. */
    /* Marriages stay marked all the time: you can only declare one on your own lead, so you need
       to see it coming rather than discover it in a one-off hint. */
    if(G.phase==='play' && meldSuitsInHand.indexOf(card.suit)!==-1 &&
       (card.rank==='K'||card.rank==='Q')){
      wrap.classList.add('meldcard');
    }
    /* Only while the deal is being shown - once play starts the glow just clutters the hand. */
    if(G.phase==='giftreveal' && G.receivedCards && G.receivedCards[G.focusSeat] &&
       cid(G.receivedCards[G.focusSeat])===cid(card)){
      wrap.classList.add('received');
    }
    /* Raise it only once the class is known, otherwise a chosen card stays buried in the fan. */
    if(wrap.classList.contains('picked')) wrap.style.zIndex='200';
    hz.appendChild(wrap);
  });

}

function meldOptionsForHand(hand){
  const res=[];
  for(const s of SUITS){
    if(hand.some(c=>c.suit===s&&c.rank==='K') && hand.some(c=>c.suit===s&&c.rank==='Q')) res.push(s);
  }
  return res;
}

/* The current bid was only mentioned in passing, so players lost track of where the auction stood
   and whether their own tap had registered. Show it as a standing line instead. */
function bidStatusLine(){
  const d=document.createElement('div');
  d.className='hint';
  d.style.cssText='font-size:14px; color:#E4C777; margin-bottom:4px;';
  if(!G.bidding || G.bidding.lastBidder===null){
    d.textContent='Licytacja otwarta — nikt jeszcze nie podbił';
  } else {
    d.textContent='Licytacja: '+G.bidding.current+' — '+seatName(G.bidding.lastBidder);
  }
  return d;
}

/* One status line, one place. Previously a floating bubble and a bottom hint could show different
   things at the same moment, which read as chaos. */
function statusText(){
  if(!G) return '';
  if(G.phase==='bidding' && G.bidding && G.bidding.turn!==G.focusSeat){
    /* The player on the musik is committed to 100 and is only asked again if somebody raises,
       which without an explanation looked like being skipped. */
    if(G.bidding.openerSeat===G.focusSeat && G.bidding.lastBidder===G.focusSeat){
      return 'Jesteś na musiku — grasz co najmniej 100. Czekam na: '+seatName(G.bidding.turn);
    }
  }
  if(linkWaitUntil){
    const left=Math.max(0, Math.ceil((linkWaitUntil-Date.now())/1000));
    return '⏳ Połączenie — czekam na: '+peerName()+' ('+left+' s)';
  }
  /* If this device has already confirmed, that is the only thing worth saying. */
  const waitStage=(G.phase==='giftreveal'||G.phase==='summary');
  if(waitStage && myAckPhase===G.phase){
    return '⏳ Czekam na: '+waitingOnNames();
  }
  if(G.message) return G.message;
  if(G.actor){
    const mine=(G.actor.seat===G.focusSeat);
    if(mine) return G.actor.lead ? 'Wychodzisz — Twoja kolej' : 'Twoja kolej';
    return seatName(G.actor.seat)+(G.actor.lead? ' wychodzi…' : ' zagrywa…');
  }
  return '';
}
/* Names, not roles, and no guessing at Polish declension: "Czekam na: Radek". */
function waitingOnNames(){
  const names=[];
  for(let s=0;s<nSeats();s++){
    if(s===G.focusSeat) continue;
    if(G.phase==='giftreveal' && s===G.bidder) continue;
    if(seatController(s)==='bot') continue;
    names.push(seatName(s));
  }
  return names.length? names.join(', ') : 'pozostałych';
}

function renderStatus(container){
  /* Your own turn gets a badge rather than just a sentence - at a glance you should know whether
     the game is waiting on you. */
  const act=activeSeatNow();
  if(G && act===G.focusSeat && (G.phase==='bidding'||G.phase==='play'||G.phase==='musikpick'||G.phase==='exchange'||G.phase==='declare')){
    const y=document.createElement('div'); y.className='yourturn';
    y.textContent='▶ Twoja kolej';
    container.appendChild(y);
  }
  const txt=statusText();
  const meldTxt=meldHintText();
  if(!txt && !meldTxt) return;
  const d=document.createElement('div'); d.className='statusbar';
  if(txt){
    const p=document.createElement('span'); p.className='primary'; p.textContent=txt;
    d.appendChild(p);
  }
  if(meldTxt){
    const m=document.createElement('span'); m.className='meld'; m.textContent=meldTxt;
    d.appendChild(m);
  }
  container.appendChild(d);
}
function meldHintText(){
  if(!G || G.phase!=='play') return '';
  const hand=G.hands[G.focusSeat]||[];
  const opts=meldOptionsForHand(hand);
  if(!opts.length) return '';
  const list=opts.map(s=>SUIT_SYMBOL[s]+' +'+MELD_VALUE[s]).join('  ');
  const myLead = (currentTurnSeat()===G.focusSeat && G.trick.length===0);
  return myLead
    ? ('Meldunek: '+list+' — wyjdź Królem lub Damą, żeby go zgłosić')
    : ('Masz meldunek: '+list+' — zgłosisz go tylko przy swoim wyjściu');
}

/* Only appears once the wait is long enough to feel broken, so it never nags during a normal
   half-second handover. */

  b.textContent='Wyślij potwierdzenie ponownie';
  b.onclick=()=>{ respond({ack:G.phase}); ackWaitSince=Date.now(); render(); };
  wrap.appendChild(b);
  return wrap;
}

function toggleExchangeCard(card){
  const ex=G.exchange;
  if(!ex.selected) ex.selected=[];
  const idx=ex.selected.findIndex(c=>cid(c)===cid(card));
  if(idx>=0){
    ex.selected.splice(idx,1);
  } else if(ex.selected.length<2){
    ex.selected.push(card);
  } else {
    /* Two already chosen: swap out the oldest instead of ignoring the tap, so the cards never
       feel stuck. */
    ex.selected.shift();
    ex.selected.push(card);
  }
  /* Guard against a card sneaking in twice if a state refresh raced with a tap. */
  const seen=new Set();
  ex.selected=ex.selected.filter(x=>{ const k=cid(x); if(seen.has(k)) return false; seen.add(k); return true; });
  render();
}

  dealCards();

  G.trickPoints=Array(nSeats()).fill(0); G.meldPoints=Array(nSeats()).fill(0); G.meldLog=[]; G.history=[];
  G.receivedCards={}; G.musikCards=null; G.summaryRows=null;
  myAckPhase=null; pendingAction=null; ackedThisHand=null; ackWaitSince=0;
  localActionQueue=[]; playerResolve=null; cardSendLock=false;
  if(NET && NET.role==='host') NET.actionQueue={};
  G.actor=null;
  G.message='Rozdano karty.';
  G.phase='bidding';
  if(G.mode==='network'){
    G.focusSeat = G.activePlayers.indexOf(0); // host is always real player 0
  } else {
    G.focusSeat=0;
  }
  if(G.mode==='hotseat') G.revealedSeat=null;
  render();
  await sleep(500);

  const result = await runBidding();
  if(result.bomb){
    await resolveBomb(result.seat);
    return;
  }
  G.bidder=result.bidder; G.bidAmount=result.amount;
  /* Show what the auction actually did, so a one-move auction does not look like nothing
     happened at all. */
  const recap=(G.bidding && G.bidding.log ? G.bidding.log.join('  ·  ') : '');
  G.message=seatName(result.bidder)+' wygrał licytację za '+result.amount
            + (recap ? ('  —  '+recap) : '');
  render();
  await sleep(1600);

  await declareContract(result.bidder);

  G.leader=result.bidder;
  G.phase='play';
  G.receivedCards={};
  G.message='';
  render();

  const trickCount = nSeats()===2 ? 10 : 8;
  for(let t=0;t<trickCount;t++){
    await playTrick();
  }

  await showSummary();
}

function dealCards(){
  const n=nSeats();
  if(G.mode!=='bots'){
    G.activePlayers=Array.from({length:n},(_,k)=>(G.dealer+1+k)%G.playerCount);
  } else {
    G.activePlayers=allSeats();
  }
  const deck=shuffle(createDeck());
  if(n===2){
    /* Two-handed deal: ten cards each and two face-down musiks of two cards. */
    G.hands[0]=sortHand(deck.slice(0,10));
    G.hands[1]=sortHand(deck.slice(10,20));
    G.musikPair=[deck.slice(20,22), deck.slice(22,24)];
    G.musik=null;
  } else {
    G.hands[0]=sortHand(deck.slice(0,7));
    G.hands[1]=sortHand(deck.slice(7,14));
    G.hands[2]=sortHand(deck.slice(14,21));
    G.musik=deck.slice(21,24);
    G.musikPair=null;
  }
}

/* ---- Bidding ---- */
function evaluateHandStrength(hand){
  let s=0;
  for(const c of hand){
    if(c.rank==='A') s+=30;
    else if(c.rank==='10') s+=15;
    else if(c.rank==='K') s+=8;
    else if(c.rank==='Q') s+=6;
    else if(c.rank==='J') s+=3;
  }
  for(const suit of SUITS){
    if(hand.some(c=>c.suit===suit&&c.rank==='K') && hand.some(c=>c.suit===suit&&c.rank==='Q')){
      s += MELD_VALUE[suit]*0.55;
    }
  }
  const counts={};
  for(const c of hand) counts[c.suit]=(counts[c.suit]||0)+1;
  const maxSuit=Math.max(...Object.values(counts));
  s += (maxSuit-1)*10;
  return s;
}

async function runBidding(){
  /* The player to the dealer's left is on the musik and is committed to 100 without having to
     tap it, exactly as at a real table. The auction therefore opens with the next player, who
     can raise to 110, pass, or bomb. */
  const opener = (G.mode==='hotseat'||G.mode==='network') ? 0 : (G.dealer+1)%nSeats();
  const b={
    current:100, lastBidder:opener, openerSeat:opener, passed:Array(nSeats()).fill(false),
    turn:(opener+1)%nSeats(), log:[seatName(opener)+': 100 (z musiku)']
  };
  G.bidding=b;
  while(true){
    render();
    const seat=b.turn;
    if(!b.passed[seat]){
      let action;
      const ctrl=seatController(seat);
      if(ctrl==='remote'){
        action = await waitForRemote(seat);
      } else if(ctrl==='local'){
        await ensureRevealed(seat);
        render();
        action = await waitForPlayer();
        releaseReveal();
      } else {
        await sleep(500+Math.random()*400);
        const strength=evaluateHandStrength(G.hands[seat]);
        const threshold=round10(90+strength+(Math.random()*20-10));
        const next=b.current+10;
        const hardCap=maxBidFor(G.hands[seat]);
        const rid=seatRealId(seat);
        const bombsLeft = G.bombsEnabled ? (BOMB_LIMIT-((G.bombsUsed&&G.bombsUsed[rid])||0)) : 0;
        /* A hopeless hand facing a high contract is exactly what a bomb is for. */
        if(bombsLeft>0 && strength<45 && b.current>=120 && Math.random()<0.5){
          action={type:'bomb'};
        } else {
          action = (next<=threshold && next<=hardCap) ? {type:'bid', amount:next} : {type:'pass'};
        }
      }
      if(action.type==='bomb'){
        return {bomb:true, seat};
      } else if(action.type==='bid'){
        b.current=action.amount; b.lastBidder=seat;
        b.log.push(seatName(seat)+': '+action.amount);
      } else {
        b.passed[seat]=true;
        b.log.push(seatName(seat)+': pas');
      }
    }
    const active=allSeats().filter(i=>!b.passed[i]);
    if(active.length===1){
      const bidder=active[0];
      const amount = b.lastBidder!==null ? b.current : 100;
      return {bidder, amount};
    }
    do{ b.turn=(b.turn+1)%nSeats(); } while(b.passed[b.turn]);
  }
}

function waitForPlayer(){
  if(localActionQueue.length){ return Promise.resolve(localActionQueue.shift()); }
  return new Promise(res=>{ playerResolve=res; });
}

/* ---- Exchange ---- */
/* Everyone should actually see what the musik held and which card they were handed, so the deal is
   staged: reveal the three cards, wait for every human to confirm, do the exchange, then show each
   opponent the card they received (highlighted in their hand) and wait again before play starts. */
async function ackAll(skipSeat){
  const stage=G.phase;
  /* Ignore a duplicate or late confirmation meant for an earlier stage. */
  async function remoteAck(s){
    for(let i=0;i<6;i++){
      const a=await waitForRemote(s);
      if(!a || !a.ack || a.ack===stage) return a;
    }
    return null;
  }
  const remotePromises=[];
  for(let s=0;s<nSeats();s++){
    if(s===skipSeat) continue;
    if(seatController(s)==='remote') remotePromises.push(remoteAck(s));
  }
  for(let s=0;s<nSeats();s++){
    if(s===skipSeat) continue;
    if(seatController(s)==='local'){
      await ensureRevealed(s);
      render();
      await waitForPlayer();
      releaseReveal();
    }
  }
  if(remotePromises.length){
    /* Last-resort guard: rather than hanging in silence, say something and let play continue. */
    let warned=false;
    const warn=setTimeout(()=>{
      warned=true;
      if(G){ G.message='⏳ To trwa dłużej niż zwykle — czekam na: '+waitingOnNames(); try{ render(); }catch(e){} }
    }, 12000);
    await Promise.race([
      Promise.all(remotePromises),
      new Promise(res=>setTimeout(res, 30000))
    ]);
    clearTimeout(warn);
    if(warned && G){ G.message=''; }
  }
}

/* Two-handed exchange: the bidder opens one of the two musiks, takes those two cards, then puts
   two of their own back face down. Those buried cards score for the bidder at the end. */
async function doExchangeTwo(bidder){
  const bidderCtrl=seatController(bidder);
  G.receivedCards={};
  G.phase='musikpick';
  G.message=seatName(bidder)+' wybiera musik.';
  if(bidderCtrl==='local'){ G.focusSeat=bidder; }
  render();

  let pick;
  if(bidderCtrl==='remote'){
    pick=await waitForRemote(bidder);
  } else if(bidderCtrl==='local'){
    await ensureRevealed(bidder);
    render();
    pick=await waitForPlayer();
    releaseReveal();
  } else {
    pick={musik: Math.random()<0.5 ? 0 : 1};
  }
  const chosen = (pick && pick.musik===1) ? 1 : 0;
  G.chosenMusik=chosen;
  G.musikCards=G.musikPair[chosen].slice();
  G.hands[bidder]=sortHand(G.hands[bidder].concat(G.musikCards));

  G.phase='musikreveal';
  G.message='Odkryty musik '+(chosen+1)+' — '+seatName(bidder)+' bierze te karty.';
  render();
  await sleep(2200);

  /* Bidder buries two cards. */
  let buried;
  if(bidderCtrl==='remote'){
    G.phase='exchange';
    G.exchange={step:'select', selected:[], assign:{}};
    render();
    const res=await waitForRemote(bidder);
    buried=res.selected;
  } else if(bidderCtrl==='local'){
    await ensureRevealed(bidder);
    G.phase='exchange';
    G.exchange={step:'select', selected:[], assign:{}};
    render();
    const res=await waitForPlayer();
    buried=res.selected;
    releaseReveal();
  } else {
    const hand=[...G.hands[bidder]];
    hand.sort((a,b)=>POINTS[a.rank]-POINTS[b.rank] || rankIdx(a.rank)-rankIdx(b.rank));
    buried=[hand[0],hand[1]];
  }
  buried.forEach(card=>{
    G.hands[bidder]=G.hands[bidder].filter(x=>cid(x)!==cid(card));
  });
  /* Buried cards leave the deal entirely; counting them for the bidder made the score look wrong. */

  G.musikCards=null;
  G.message='';
  render();
}

/* After seeing the musik the hand can be much stronger than the auction suggested, so the player
   declares the final contract - never less than they bid, but as much more as they dare. */
async function declareContract(bidder){
  const ctrl=seatController(bidder);
  G.phase='declare';
  G.message='';
  G.actor=null;
  if(ctrl==='local'){ G.focusSeat=bidder; }
  render();

  let chosen=G.bidAmount;
  if(ctrl==='remote'){
    const res=await waitForRemote(bidder);
    chosen=(res && res.amount) ? res.amount : G.bidAmount;
  } else if(ctrl==='local'){
    await ensureRevealed(bidder);
    render();
    const res=await waitForPlayer();
    chosen=(res && res.amount) ? res.amount : G.bidAmount;
    releaseReveal();
  } else {
    const strength=evaluateHandStrength(G.hands[bidder]);
    const target=round10(90+strength);
    chosen=Math.min(Math.max(G.bidAmount, target), maxBidFor(G.hands[bidder]));
  }
  G.bidAmount=Math.max(G.bidAmount, Math.min(chosen, maxBidFor(G.hands[bidder])));
  G.message='';
  render();
}

async function doExchange(bidder){
  if(nSeats()===2){ await doExchangeTwo(bidder); return; }
  G.musikCards=G.musik.slice();               /* kept for display after they join the hand */
  G.hands[bidder]=sortHand(G.hands[bidder].concat(G.musik));
  G.receivedCards={};
  G.phase='musikreveal';
  G.message=seatName(bidder)+' wygrał licytację i bierze musik.';
  const bidderCtrl=seatController(bidder);
  if(bidderCtrl==='local'){ G.focusSeat=bidder; }
  render();
  await sleep(2200);

  let gifts;
  if(bidderCtrl==='remote'){
    G.phase='exchange';
    G.exchange={step:'select', selected:[], assign:{}};
    render();
    const result = await waitForRemote(bidder);
    gifts=[ {to:result.assign[0], card:result.selected[0]}, {to:result.assign[1], card:result.selected[1]} ];
  } else if(bidderCtrl==='local'){
    await ensureRevealed(bidder);
    G.phase='exchange';
    G.exchange={step:'select', selected:[], assign:{}};
    render();
    const result = await waitForPlayer();
    gifts=[ {to:result.assign[0], card:result.selected[0]}, {to:result.assign[1], card:result.selected[1]} ];
    releaseReveal();
  } else {
    const hand=[...G.hands[bidder]];
    const weight=c=>{
      const isMeldPart = hand.some(x=>x.suit===c.suit && ((x.rank==='K'&&c.rank==='Q')||(x.rank==='Q'&&c.rank==='K')));
      return (isMeldPart?1000:0) + POINTS[c.rank]*10 + rankIdx(c.rank);
    };
    hand.sort((a,b)=>weight(a)-weight(b));
    const give=[hand[0],hand[1]];
    const opponents=allSeats().filter(i=>i!==bidder);
    gifts=[{to:opponents[0],card:give[0]},{to:opponents[1],card:give[1]}];
  }

  gifts.forEach(g=>{
    G.hands[bidder]=G.hands[bidder].filter(c=>cid(c)!==cid(g.card));
    G.hands[g.to]=sortHand(G.hands[g.to].concat([g.card]));
    G.receivedCards[g.to]=g.card;
  });
  G.musikCards=null;

  G.phase='giftreveal';
  G.message=seatName(bidder)+' rozdał karty. Sprawdź, co dostałeś.';
  render();
  await ackAll(bidder);

  G.musikCards=null;
  G.message='';
  render();
}

/* ---- Trick play ---- */
function legalMoves(seat){
  const hand=G.hands[seat];
  if(G.trick.length===0) return hand.slice();
  const ledSuit=G.trick[0].card.suit;
  const trump=G.trump;

  /* Would this card currently take the trick? Asking the real winner logic keeps the rule correct
     in every case, including when trump itself was led (where the old check wrongly skipped the
     obligation to overtrump). */
  function wouldWin(card){
    const sim=[...G.trick, {seat, card}];
    return trickWinnerIndex(sim, ledSuit, trump) === sim.length-1;
  }

  /* Follow suit first; among the led suit, beat the trick if you can. */
  const haveLed=hand.filter(c=>c.suit===ledSuit);
  if(haveLed.length>0){
    const winning=haveLed.filter(wouldWin);
    return winning.length ? winning : haveLed;
  }
  /* Cannot follow: must trump if holding trump, and must overtrump when possible. */
  if(trump){
    const haveTrump=hand.filter(c=>c.suit===trump);
    if(haveTrump.length>0){
      const winning=haveTrump.filter(wouldWin);
      return winning.length ? winning : haveTrump;
    }
  }
  return hand.slice();
}

function trickWinnerIndex(trick, ledSuit, trump){
  let bestIdx=0;
  for(let i=1;i<trick.length;i++){
    const cur=trick[bestIdx].card, cand=trick[i].card;
    const curIsTrump = trump && cur.suit===trump;
    const candIsTrump = trump && cand.suit===trump;
    if(candIsTrump && !curIsTrump){ bestIdx=i; continue; }
    if(candIsTrump && curIsTrump){ if(rankIdx(cand.rank)>rankIdx(cur.rank)) bestIdx=i; continue; }
    if(!candIsTrump && curIsTrump){ continue; }
    if(cand.suit===ledSuit && cur.suit===ledSuit){ if(rankIdx(cand.rank)>rankIdx(cur.rank)) bestIdx=i; continue; }
    if(cand.suit===ledSuit && cur.suit!==ledSuit){ bestIdx=i; continue; }
  }
  return bestIdx;
}

async function playTrick(){
  G.trick=[];
  render();
  const order=allSeats().map((_,k)=>(G.leader+k)%nSeats());
  for(const seat of order){
    render();
    let card;
    const ctrl=seatController(seat);
    if(ctrl==='remote'){
      G.message='';
      G.actor={seat, lead:(G.trick.length===0)};
      render();
      card = await awaitValidCard(seat, ()=>waitForRemote(seat));
    } else if(ctrl==='local'){
      await ensureRevealed(seat);
      G.message='';
      G.actor={seat, lead:(G.trick.length===0)};
      render();
      card = await awaitValidCard(seat, ()=>waitForPlayer());
      releaseReveal();
    } else {
      await sleep(550+Math.random()*450);
      card = chooseBotCard(seat);
    }
    playCardForSeat(seat, card);
    render();
    await sleep(250);
  }
  const ledSuit=G.trick[0].card.suit;
  const wIdx=trickWinnerIndex(G.trick, ledSuit, G.trump);
  const winnerSeat=G.trick[wIdx].seat;
  const pts=G.trick.reduce((s,t)=>s+POINTS[t.card.rank],0);
  G.trickPoints[winnerSeat]+=pts;
  if(!G.history) G.history=[];
  G.history.push({
    cards: G.trick.map(t=>({seat:t.seat, card:t.card})),
    winner: winnerSeat,
    pts,
    no: G.tricksPlayed+1
  });
  G.actor=null;
  G.message=seatName(winnerSeat)+' bierze lewę (+'+pts+' pkt)';
  G.leader=winnerSeat;
  G.tricksPlayed++;
  if(seatController(winnerSeat)==='local'){ G.focusSeat=winnerSeat; }
  render();
  await sleep(1400);
}

/* A duplicate or stale card action (a double tap, or one replayed from the queue) used to be
   pushed onto the table even though the card had already left the hand. That desynchronised the
   hands and let the same ace appear again and again. Nothing reaches the table unvalidated now. */
function isPlayableNow(seat, card){
  if(!card || !card.suit || !card.rank) return false;
  if(!G.hands[seat].some(x=>cid(x)===cid(card))) return false;
  return legalMoves(seat).some(x=>cid(x)===cid(card));
}
async function awaitValidCard(seat, getter){
  for(let attempt=0; attempt<25; attempt++){
    const card=await getter();
    if(isPlayableNow(seat, card)) return card;
    /* silently drop it and keep waiting for a real move */
  }
  const legal=legalMoves(seat);
  return legal[0];
}

function playCardForSeat(seat, card){
  const isLead = G.trick.length===0;
  const hand=G.hands[seat];
  if(isLead && (card.rank==='K'||card.rank==='Q')){
    const partnerRank = card.rank==='K'?'Q':'K';
    const hasPartner = hand.some(c=>c.suit===card.suit && c.rank===partnerRank && cid(c)!==cid(card));
    if(hasPartner){
      const val=MELD_VALUE[card.suit];
      G.meldPoints[seat]+=val;
      let trumpMsg='';
      if(!G.trump || val>MELD_VALUE[G.trump]){
        G.trump=card.suit;
        trumpMsg=' Nowy atu: '+SUIT_SYMBOL[card.suit]+'.';
      }
      G.meldLog.push(seatName(seat)+': meldunek '+SUIT_SYMBOL[card.suit]+' +'+val);
      G.message=seatName(seat)+' melduje '+SUIT_SYMBOL[card.suit]+' (+'+val+' pkt).'+trumpMsg;
    }
  }
  G.hands[seat]=hand.filter(c=>cid(c)!==cid(card));
  G.trick.push({seat, card});
}

/* Cards nobody can still hold: everything already played plus everything in our own hand. Knowing
   this is what lets a bot tell a winning ace from a losing one. */
function cardsSeenBy(seat){
  const seen=new Set();
  (G.history||[]).forEach(h=>h.cards.forEach(t=>seen.add(cid(t.card))));
  (G.trick||[]).forEach(t=>seen.add(cid(t.card)));
  (G.hands[seat]||[]).forEach(cd=>seen.add(cid(cd)));
  return seen;
}
/* Is this the best card still outstanding in its suit? */
function isTopRemaining(card, seen){
  for(let i=rankIdx(card.rank)+1;i<RANKS.length;i++){
    if(!seen.has(card.suit+'-'+RANKS[i])) return false;
  }
  return true;
}
function trickValue(){
  return (G.trick||[]).reduce((s,t)=>s+POINTS[t.card.rank],0);
}
function countSuit(hand, suit){ return hand.filter(c=>c.suit===suit).length; }

function chooseBotCard(seat){
  const legal=legalMoves(seat);
  if(legal.length===1) return legal[0];

  /* Bots are meant to feel like opponents, not solvers, so now and then they simply pick something
     reasonable-but-wrong, the way a distracted human would. */
  if(Math.random() < BOT_MISTAKE_RATE){
    return legal[Math.floor(Math.random()*legal.length)];
  }

  const hand=G.hands[seat];
  const seen=cardsSeenBy(seat);
  const isLead=G.trick.length===0;
  const trump=G.trump;

  if(isLead){
    /* Declaring a marriage is almost always worth it; take the biggest one available. */
    const melds=meldOptionsForHand(hand).filter(s=>legal.some(cd=>cd.suit===s));
    if(melds.length){
      melds.sort((a,b)=>MELD_VALUE[b]-MELD_VALUE[a]);
      const suit=melds[0];
      return legal.find(cd=>cd.suit===suit&&cd.rank==='K') || legal.find(cd=>cd.suit===suit&&cd.rank==='Q');
    }
    /* Cash a guaranteed winner while it is still good. */
    const sureWinners=legal.filter(cd=>isTopRemaining(cd,seen) && (!trump || cd.suit===trump || countSuit(hand,trump)===0 || true));
    const valuable=sureWinners.filter(cd=>POINTS[cd.rank]>=10);
    if(valuable.length){
      valuable.sort((a,b)=>POINTS[b.rank]-POINTS[a.rank]);
      return valuable[0];
    }
    /* Otherwise probe with a cheap card from the longest non-trump suit, keeping trumps back. */
    const suits=SUITS.filter(s=>legal.some(cd=>cd.suit===s));
    suits.sort((a,b)=>{
      const at=(a===trump)?1:0, bt=(b===trump)?1:0;
      if(at!==bt) return at-bt;
      return countSuit(hand,b)-countSuit(hand,a);
    });
    const pick=legal.filter(cd=>cd.suit===suits[0]);
    pick.sort((a,b)=>POINTS[a.rank]-POINTS[b.rank] || rankIdx(a.rank)-rankIdx(b.rank));
    return pick[0];
  }

  /* Following: work out which of our legal cards actually takes the trick. */
  const ledSuit=G.trick[0].card.suit;
  const winners=legal.filter(cd=>{
    const sim=[...G.trick,{seat,card:cd}];
    return trickWinnerIndex(sim, ledSuit, trump)===sim.length-1;
  });
  const pot=trickValue();
  const isLast=(G.trick.length===nSeats()-1);

  if(winners.length){
    winners.sort((a,b)=>POINTS[a.rank]-POINTS[b.rank] || rankIdx(a.rank)-rankIdx(b.rank));
    const cheapest=winners[0];
    /* Spending an ace to win an empty trick is how points get thrown away, so only pay when the
       pot is worth it, when the win is certain, or when we are last and it is free. */
    const worthIt = pot>=8 || POINTS[cheapest.rank]<=4 || isLast;
    if(worthIt) return cheapest;
  }

  /* Cannot or should not win: shed the least useful card, preferring to void a short side suit
     rather than break up length or waste trumps. */
  const dump=legal.slice().sort((a,b)=>{
    if(POINTS[a.rank]!==POINTS[b.rank]) return POINTS[a.rank]-POINTS[b.rank];
    const at=(a.suit===trump)?1:0, bt=(b.suit===trump)?1:0;
    if(at!==bt) return at-bt;
    const al=countSuit(hand,a.suit), bl=countSuit(hand,b.suit);
    if(al!==bl) return al-bl;
    return rankIdx(a.rank)-rankIdx(b.rank);
  });
  return dump[0];
}

/* A bomb cancels the deal outright: the thrower scores nothing and everyone else takes 60.
   Each player only gets BOMB_LIMIT of them for the whole game. */
async function resolveBomb(seat){
  const rid=seatRealId(seat);
  if(!G.bombsUsed) G.bombsUsed={};
  G.bombsUsed[rid]=(G.bombsUsed[rid]||0)+1;
  const rows=[];
  for(let s=0;s<nSeats();s++){
    const r=seatRealId(s);
    const delta=(s===seat)?0:BOMB_BONUS;
    G.scores[r]+=delta;
    rows.push({name:seatName(s), tricks:0, melds:0, total:0, delta,
               note:(s===seat)?'rzucił bombę — rozdanie przepada':'za bombę przeciwnika',
               bidder:(s===seat), score:G.scores[r]});
  }
  G.summaryRows=rows;
  G.phase='summary';
  G.bidder=null; G.bidAmount=0;
  G.message='💣 '+seatName(seat)+' rzuca bombę';
  render();
  await ackAll();
}

async function showSummary(){
  const totals=allSeats().map(i=>round10(G.trickPoints[i]+G.meldPoints[i]));
  /* Structured rather than one long sentence, so every player can check the arithmetic:
     the declarer banks what they actually took when the contract is made, and loses the
     contract value when it is not. */
  const rows=[];
  for(let i=0;i<nSeats();i++){
    const rid=seatRealId(i);
    const tricks=G.trickPoints[i]||0, melds=G.meldPoints[i]||0;
    let delta, note;
    if(i===G.bidder){
      /* The declarer banks the contract they committed to, not the points they happened to take -
         taking more than you declared earns you nothing extra. */
      if(totals[i]>=G.bidAmount){ delta=G.bidAmount; note='kontrakt '+G.bidAmount+' wykonany'; }
      else { delta=-G.bidAmount; note='kontrakt '+G.bidAmount+' niewykonany'; }
    } else {
      delta=totals[i]; note='';
    }
    G.scores[rid]+=delta;
    rows.push({name:seatName(i), tricks, melds, total:totals[i], delta, note,
               bidder:(i===G.bidder), score:G.scores[rid]});
  }
  G.summaryRows=rows;
  G.phase='summary';
  G.message='';
  render();
  /* Both players confirm before the next deal, so nobody's tap is silently ignored. */
  await ackAll();
}

/* Keep ALL local-network (typ host) candidates - these are the ones that actually work on a shared
   Wi-Fi/hotspot with no internet. Dropping to a single candidate made connections die right after
   handshake. srflx/relay candidates are useless offline, so those are the ones we discard. */
/* ---- Compact connection token ----
   A raw SDP is ~1000+ chars, which makes a dense, hard-to-scan QR. Everything except the
   ice-ufrag, ice-pwd, DTLS fingerprint and the local-network candidates is boilerplate we can
   rebuild identically on the other side, so only those parts travel. Result: ~120 chars. */
function sdpField(sdp, prefix){
  const line = sdp.split(/\r\n|\n/).find(l=>l.indexOf(prefix)===0);
  return line ? line.slice(prefix.length).trim() : '';
}
function hexToB64(hex){
  const clean=hex.replace(/:/g,'');
  let bin='';
  for(let i=0;i<clean.length;i+=2) bin+=String.fromCharCode(parseInt(clean.substr(i,2),16));
  return btoa(bin).replace(/\+/g,'-').replace(/\//g,'_').replace(/=+$/,'');
}
function b64ToHex(b64){
  const bin=atob(b64.replace(/-/g,'+').replace(/_/g,'/'));
  const parts=[];
  for(let i=0;i<bin.length;i++) parts.push(bin.charCodeAt(i).toString(16).padStart(2,'0').toUpperCase());
  return parts.join(':');
}

function encodeSDP(desc){
  const sdp=desc.sdp;
  const ufrag=sdpField(sdp,'a=ice-ufrag:');
  const pwd=sdpField(sdp,'a=ice-pwd:');
  const fpLine=sdpField(sdp,'a=fingerprint:sha-256 ');
  const fp=hexToB64(fpLine);
  const cands=[];
  sdp.split(/\r\n|\n/).forEach(l=>{
    if(l.indexOf('a=candidate:')===0 && l.indexOf('typ host')!==-1){
      const p=l.slice('a=candidate:'.length).split(' ');
      /* foundation component transport priority ip port typ host */
      if(p.length>=8 && p[2].toLowerCase()==='udp'){
        cands.push(p[4]+':'+p[5]);
      }
    }
  });
  const uniq=[...new Set(cands)].slice(0,4);
  if(uniq.length===0){
    alert('Uwaga: ten telefon nie znalazł żadnego adresu w sieci lokalnej. Upewnij się, że oba telefony są na tym samym hotspocie/Wi-Fi, i spróbuj ponownie.');
  }
  return [(desc.type==='offer'?'O':'A'), ufrag, pwd, fp, uniq.join(',')].join('~');
}

/* Input may arrive as a bare token, or as the full link the QR now carries. Accept either. */
function extractToken(text){
  if(!text) return '';
  let t=String(text).trim();
  const hash=t.indexOf('#j=')!==-1 ? t.indexOf('#j=') : t.indexOf('#a=');
  if(hash!==-1) t=t.slice(hash+3);
  return t.replace(/\s+/g,'');
}

function decodeSDP(text){
  const raw=extractToken(text);
  if(raw.indexOf('~')===-1) throw new Error('nieprawidłowy kod');
  const [kind, ufrag, pwd, fp, candStr] = raw.split('~');
  const type = kind==='O' ? 'offer' : 'answer';
  const fingerprint = b64ToHex(fp);
  const candLines = (candStr||'').split(',').filter(Boolean).map((cp,i)=>{
    const idx=cp.lastIndexOf(':');
    const ip=cp.slice(0,idx), port=cp.slice(idx+1);
    return 'a=candidate:'+(i+1)+' 1 udp '+(2113937151-i)+' '+ip+' '+port+' typ host';
  });
  const sdp=[
    'v=0',
    'o=- 0 2 IN IP4 127.0.0.1',
    's=-',
    't=0 0',
    'a=group:BUNDLE 0',
    'a=extmap-allow-mixed',
    'a=msid-semantic: WMS',
    'm=application 9 UDP/DTLS/SCTP webrtc-datachannel',
    'c=IN IP4 0.0.0.0',
    'a=ice-ufrag:'+ufrag,
    'a=ice-pwd:'+pwd,
    'a=ice-options:trickle',
    'a=fingerprint:sha-256 '+fingerprint,
    'a=setup:'+(type==='offer'?'actpass':'active'),
    'a=mid:0',
    'a=sctp-port:5000',
    'a=max-message-size:262144'
  ].concat(candLines).join('\r\n')+'\r\n';
  return {type, sdp};
}

/* Decode a QR from a still image bitmap. Native BarcodeDetector first (far more robust), jsQR as fallback. */
async function decodeQRFromCanvas(canvas, ctx){
  if('BarcodeDetector' in window){
    try{
      const det=new BarcodeDetector({formats:['qr_code']});
      const found=await det.detect(canvas);
      if(found && found.length && found[0].rawValue) return found[0].rawValue;
    }catch(e){ /* fall through to jsQR */ }
  }
  if(typeof jsQR==='undefined') return null; /* library missing - caller reports this explicitly */
  try{
    const imgData=ctx.getImageData(0,0,canvas.width,canvas.height);
    const code=jsQR(imgData.data, imgData.width, imgData.height, {inversionAttempts:'attemptBoth'});
    if(code && code.data) return code.data;
  }catch(e){}
  return null;
}

function qrReaderAvailable(){
  return ('BarcodeDetector' in window) || (typeof jsQR!=='undefined');
}

/* Uses the phone's OWN camera app to take a photo, then decodes the still.
   This sidesteps the in-browser live preview entirely - the native camera focuses properly. */
function scanQRByPhoto(){
  return new Promise((resolve,reject)=>{
    const input=document.createElement('input');
    input.type='file';
    input.accept='image/*';
    input.setAttribute('capture','environment');
    input.style.display='none';
    document.body.appendChild(input);
    input.onchange=async ()=>{
      const file=input.files && input.files[0];
      document.body.removeChild(input);
      if(!file){ reject('cancelled'); return; }
      try{
        const bitmap=await createImageBitmap(file);
        const full=document.createElement('canvas');
        full.width=bitmap.width; full.height=bitmap.height;
        full.getContext('2d',{willReadFrequently:true}).drawImage(bitmap,0,0);

        /* Try a whole ladder of renderings. A phone photo is far larger than jsQR likes, and the
           code usually sits in the middle of a wide frame, so both downscaling and centre-cropping
           matter. First hit wins. */
        async function attempt(sx, sy, sw, sh, target, label){
          const c=document.createElement('canvas');
          const scale=Math.min(1, target/Math.max(sw,sh));
          c.width=Math.max(1,Math.round(sw*scale));
          c.height=Math.max(1,Math.round(sh*scale));
          const x=c.getContext('2d',{willReadFrequently:true});
          x.drawImage(full, sx, sy, sw, sh, 0, 0, c.width, c.height);
          const r=await decodeQRFromCanvas(c, x);
          return r;
        }

        const W=full.width, H=full.height;
        const plans=[];
        [1600, 1100, 800, 2400].forEach(t=>plans.push([0,0,W,H,t,'całość@'+t]));
        /* centre crops at a few zoom levels */
        [0.7, 0.5, 0.35].forEach(f=>{
          const cw=Math.round(W*f), ch=Math.round(H*f);
          const sx=Math.round((W-cw)/2), sy=Math.round((H-ch)/2);
          plans.push([sx,sy,cw,ch,1100,'środek '+Math.round(f*100)+'%']);
        });

        let data=null;
        for(const p of plans){
          data=await attempt(p[0],p[1],p[2],p[3],p[4],p[5]);
          if(data) break;
        }
        if(data) resolve(data);
        else reject('not-found');
      }catch(e){ reject(e); }
    };
    input.click();
  });
}

async function scanQR(promptText){
  return new Promise(async (resolve,reject)=>{
    const modal=el('qrScanModal');
    el('qrScanTitle').textContent=promptText;
    el('qrScanError').textContent='';
    modal.style.display='flex';
    if(!qrReaderAvailable()){
      el('qrScanError').style.color='#c0392b';
      el('qrScanError').textContent='Czytnik kodów nie wczytał się (brak internetu przy pierwszym uruchomieniu). Otwórz aplikację raz z internetem, albo użyj kodu tekstowego.';
    }
    const video=el('qrVideo');
    const canvas=document.createElement('canvas');
    const ctx=canvas.getContext('2d', {willReadFrequently:true});
    let stopped=false;
    let stream=null;

    function cleanup(){
      stopped=true;
      if(stream){ try{ stream.getTracks().forEach(t=>t.stop()); }catch(e){} }
      modal.style.display='none';
    }
    el('qrScanCancel').onclick=()=>{ cleanup(); reject('cancelled'); };
    el('qrPhotoBtn').onclick=async ()=>{
      const wasStopped=stopped;
      stopped=true;
      if(stream){ try{ stream.getTracks().forEach(t=>t.stop()); }catch(e){} }
      try{
        const data=await scanQRByPhoto();
        modal.style.display='none';
        resolve(data);
      }catch(e){
        el('qrScanError').style.color='#c0392b';
        el('qrScanError').textContent = e==='not-found'
          ? 'Nie odczytano kodu ze zdjęcia. Zrób ostrzejsze zdjęcie z bliska, tak by kod wypełniał kadr.'
          : 'Nie udało się zrobić zdjęcia. Spróbuj ponownie albo użyj kodu tekstowego.';
      }
    };

    const constraintAttempts=[
      {video:{facingMode:{ideal:'environment'}, width:{ideal:1920}, height:{ideal:1080}, advanced:[{focusMode:'continuous'}]}},
      {video:{facingMode:{ideal:'environment'}, width:{ideal:1280}, height:{ideal:720}}},
      {video:true}
    ];
    for(const cst of constraintAttempts){
      try{ stream=await navigator.mediaDevices.getUserMedia(cst); break; }catch(e){ /* try next */ }
    }
    if(!stream){
      el('qrScanError').style.color='#c0392b';
      el('qrScanError').textContent='Brak dostępu do aparatu na żywo — użyj przycisku „Zrób zdjęcie kodu" poniżej.';
      return;
    }
    try{
      const track=stream.getVideoTracks()[0];
      if(track && track.getCapabilities && track.applyConstraints){
        const caps=track.getCapabilities();
        if(caps.focusMode && caps.focusMode.indexOf('continuous')!==-1){
          await track.applyConstraints({advanced:[{focusMode:'continuous'}]}).catch(()=>{});
        }
      }
    }catch(e){}
    video.srcObject=stream;
    try{ await video.play(); }catch(e){}
    el('qrScanError').style.color='#7a7360';
    el('qrScanError').textContent='Szukam kodu…';
    setTimeout(()=>{
      if(!stopped){ el('qrScanError').textContent='Nie widzę kodu na żywo — najpewniej aparat nie łapie ostrości. Użyj przycisku „Zrób zdjęcie kodu" poniżej, to działa pewniej.'; }
    }, 5000);

    let busy=false;
    async function tick(){
      if(stopped) return;
      if(!busy && video.readyState===video.HAVE_ENOUGH_DATA && video.videoWidth>0){
        busy=true;
        canvas.width=video.videoWidth; canvas.height=video.videoHeight;
        ctx.drawImage(video,0,0,canvas.width,canvas.height);
        const data=await decodeQRFromCanvas(canvas, ctx);
        busy=false;
        if(data && !stopped){
          cleanup();
          resolve(data);
          return;
        }
      }
      requestAnimationFrame(tick);
    }
    requestAnimationFrame(tick);
  });
}

/* ---- Link-based pairing ----
   The phone's own camera app reads these QR codes reliably and opens the link, which is far more
   dependable than any in-page scanner. The connection token rides in the URL fragment.
   For the answer leg the host's game tab must stay alive, so the freshly opened tab hands the
   token over through localStorage, which is shared between tabs of the same site. */
const ANSWER_KEY='tysiac_answer_token';

function appBaseURL(){
  return location.origin + location.pathname;
}
/* The link-in-QR trick needs a real http(s) address. Opened straight from a file there is no
   usable URL, so fall back to the bare token, which the in-app scanner and paste box still take. */
function qrPayload(prefix, code){
  if(location.protocol==='http:' || location.protocol==='https:'){
    return appBaseURL()+'#'+prefix+'='+code;
  }
  return code;
}

function handleIncomingHash(){
  const h=location.hash||'';
  if(h.indexOf('#j=')===0){
    const token=decodeURIComponent(h.slice(3));
    history.replaceState(null,'',appBaseURL());
    openNetJoin();
    el('joinCodeInput').value=token;
    el('netJoinStatus').textContent='Odczytano kod gospodarza — generuję odpowiedź…';
    clientGenerateAnswerText(token);
    return true;
  }
  if(h.indexOf('#a=')===0){
    const token=decodeURIComponent(h.slice(3));
    history.replaceState(null,'',appBaseURL());
    /* If this very tab is the host, apply immediately. */
    if(NET && NET.role==='host' && NET.seatConn && NET.seatConn[1]){
      hostConnectFromAnswerText(token);
    } else {
      /* Otherwise pass it to the tab that actually holds the game. */
      try{ localStorage.setItem(ANSWER_KEY, JSON.stringify({token, at:Date.now()})); }catch(e){}
      document.body.innerHTML='<div style="padding:28px; font-family:-apple-system,sans-serif; color:#F4F1EA; text-align:center;">'+
        '<h2 style="font-family:Georgia,serif; color:#E4C777;">Kod odebrany</h2>'+
        '<p>Wróć do karty z grą — połączenie właśnie się kończy.</p>'+
        '<p style="color:#CFC9B8; font-size:13px;">Tę kartę możesz zamknąć.</p></div>';
    }
    return true;
  }
  return false;
}

/* The host tab watches for an answer token handed over by a camera-opened tab. */
function watchForAnswerToken(){
  let lastSeen=0;
  function tryConsume(){
    if(!(NET && NET.role==='host' && NET.seatConn && NET.seatConn[1])) return;
    if(G && G.mode==='network') return; /* already playing */
    let raw;
    try{ raw=localStorage.getItem(ANSWER_KEY); }catch(e){ return; }
    if(!raw) return;
    let obj; try{ obj=JSON.parse(raw); }catch(e){ return; }
    if(!obj || !obj.token || obj.at===lastSeen) return;
    lastSeen=obj.at;
    try{ localStorage.removeItem(ANSWER_KEY); }catch(e){}
    el('hostSetupStatus').textContent='Odebrano kod odpowiedzi — łączenie…';
    hostConnectFromAnswerText(obj.token);
  }
  window.addEventListener('storage', (e)=>{ if(e.key===ANSWER_KEY) tryConsume(); });
  setInterval(tryConsume, 1000); /* storage events are unreliable on iOS; poll as well */
}

/* Safari/Chrome hide the real local IP behind an mDNS ".local" hostname until the page has been
   granted camera/mic access. Two iPhones cannot resolve each other's .local names over a personal
   hotspot, so the connection dies seconds after handshake. Taking (and immediately releasing) a
   camera permission first makes the browser publish real 192.168.x.x candidates instead. */
async function primeLocalNetworkAccess(){
  try{
    const s=await navigator.mediaDevices.getUserMedia({video:true});
    s.getTracks().forEach(t=>t.stop());
    return true;
  }catch(e){
    return false;
  }
}

/* Backgrounding an iPhone kills the peer connection. Rather than forcing a fresh game, the host
   can publish a new pairing code while keeping the current hand intact; once the friend rejoins,

/* If the two phones report addresses on different subnets they simply cannot reach each other.
   The usual cause is a VPN or iCloud Private Relay on one phone, which reports the tunnel's
   virtual address instead of the hotspot address. Naming that beats a generic failure message. */
function subnetsOf(sdp){
  const out=[];
  (sdp||'').split(/\r\n|\n/).forEach(l=>{
    if(l.indexOf('a=candidate:')===0 && l.indexOf('typ host')!==-1){
      const ip=l.slice(12).split(' ')[4]||'';
      const m=ip.match(/^(\d+)\.(\d+)\.(\d+)\.\d+$/);
      if(m) out.push(m[1]+'.'+m[2]+'.'+m[3]);
    }
  });
  return [...new Set(out)];
}
function checkSubnetMatch(pc){
  try{
    if(!pc || !pc.localDescription || !pc.remoteDescription) return null;
    const mine=subnetsOf(pc.localDescription.sdp);
    const theirs=subnetsOf(pc.remoteDescription.sdp);
    if(!mine.length || !theirs.length) return null;
    const shared=mine.some(a=>theirs.indexOf(a)!==-1);
    return shared ? null : {mine, theirs};
  }catch(e){ return null; }
}
function subnetWarningText(info){
  return '⚠ Telefony są w różnych sieciach (Ty: '+info.mine.join(', ')+'.x, drugi: '+info.theirs.join(', ')+
         '.x). Najczęstsza przyczyna to włączony VPN lub iCloud Private Relay — wyłącz go, połącz się z hotspotem i spróbuj ponownie.';
}

/* iOS never exposes the hotspot-providing phone's own address on that network to the browser, so
   that side advertises only its cellular address and the two phones never find each other. The
   hotspot gateway is always 172.20.10.1, so when the peer is clearly on a hotspot and we are not,
   advertise that address too. It only helps if the media socket is bound to all interfaces, but
   it costs nothing to try. */
const IOS_HOTSPOT_GATEWAY='172.20.10.1';
function augmentCandidatesForHotspot(localToken, remoteSdp){
  try{
    const parts=localToken.split('~');
    const cands=(parts[4]||'').split(',').filter(Boolean);
    const peerOnHotspot=(remoteSdp||'').indexOf('172.20.10.')!==-1;
    const weHaveHotspot=cands.some(c=>c.indexOf('172.20.10.')===0);
    if(peerOnHotspot && !weHaveHotspot && cands.length){
      const port=cands[0].slice(cands[0].lastIndexOf(':')+1);
      cands.push(IOS_HOTSPOT_GATEWAY+':'+port);
      parts[4]=cands.join(',');
      return parts.join('~');
    }
  }catch(e){}
  return localToken;
}

/* iOS reports a transient "disconnected" on almost any network hiccup, and it usually recovers on
   its own within a few seconds. Warning immediately produced alarming messages mid-game while play
   continued fine. So: wait out a grace period, then try to repair the link silently over the data
   channel (an ICE restart needs signalling, and the channel itself often survives), and only give
   up and ask for a fresh pairing when that fails too. */
const LINK_GRACE_MS=7000;
let linkGraceTimer=null;
let repairInFlight=false;

function connLabel(){ return (NET && NET.role==='host') ? 'graczem' : 'gospodarzem'; }
/* Who exactly are we waiting for - players want a name, not a role. */
function peerName(){
  if(!NET) return 'przeciwnika';
  if(NET.role==='host') return (NET.seatNames && NET.seatNames[1]) || 'przeciwnika';
  return (G && G.playerNames && G.playerNames[0]) || 'gospodarza';
}
let linkWaitUntil=0;
function activePC(){
  if(!NET) return null;
  return NET.role==='host'
    ? (NET.seatConn && NET.seatConn[1] && NET.seatConn[1].pc)
    : NET.pc;
}
function activeDC(){
  if(!NET) return null;
  return NET.role==='host'
    ? (NET.seatConn && NET.seatConn[1] && NET.seatConn[1].dc)
    : NET.dc;
}
function channelUsable(){
  const dc=activeDC();
  return !!(dc && dc.readyState==='open');
}

function clearLinkGrace(){
  if(linkGraceTimer){ clearTimeout(linkGraceTimer); linkGraceTimer=null; }
  linkWaitUntil=0;
}

function declareLinkHealthy(){
  clearLinkGrace();
  if(netLinkLost){
    netLinkLost=false;
    if(G && G.mode==='network'){ G.message='✓ Połączenie wróciło.'; }
    try{ render(); }catch(e){}
  }
  netLinkLost=false;
}

function declareLinkLost(pc){
  netLinkLost=true;
  const mismatch=pc?checkSubnetMatch(pc):null;
  linkWaitUntil=0;
  const why = mismatch ? subnetWarningText(mismatch)
                       : ('⚠ Utracono połączenie — gracz: '+peerName()+'. Rozdanie jest zapamiętane.');
  if(G && G.mode==='network'){ G.message=why; }
  try{ el('hostSetupStatus').textContent=why; }catch(e){}
  try{ el('netJoinStatus').textContent=why; }catch(e){}
  try{ render(); }catch(e){}
  /* Waiting for the host to notice and press a button wastes time; if the link is really gone,
     put a fresh pairing code on screen straight away so the friend can just scan it. */
  if(NET && NET.role==='host' && G && G.mode==='network' && !mismatch){
    setTimeout(()=>{ if(netLinkLost) hostReconnect(); }, 1200);
  }
}

/* Repair without any QR: renegotiate ICE over the still-open data channel. */
async function attemptSilentRepair(){
  if(repairInFlight) return false;
  if(!(NET && NET.role==='host')) return false;   /* host drives renegotiation */
  const pc=activePC(), dc=activeDC();
  if(!pc || !dc || dc.readyState!=='open') return false;
  repairInFlight=true;
  try{
    if(G && G.mode==='network'){ G.message='⏳ Odnawiam połączenie — czekam na: '+peerName(); try{ render(); }catch(e){} }
    const offer=await pc.createOffer({iceRestart:true});
    await pc.setLocalDescription(offer);
    await waitIceComplete(pc);
    dc.send(JSON.stringify({type:'reoffer', sdp:pc.localDescription.sdp}));
    return true;
  }catch(e){
    return false;
  }finally{
    setTimeout(()=>{ repairInFlight=false; }, 4000);
  }
}

function monitorConnection(pc, label){
  if(!pc) return;
  pc.onconnectionstatechange=async ()=>{
    const st=pc.connectionState;
    if(st==='connected'){
      declareLinkHealthy();
      return;
    }
    if(st==='closed' || st==='failed'){
      clearLinkGrace();
      const repaired = channelUsable() ? await attemptSilentRepair() : false;
      if(!repaired) declareLinkLost(pc);
      return;
    }
    if(st==='disconnected'){
      /* Give it a chance to heal before saying anything to the players. */
      clearLinkGrace();
      linkWaitUntil=Date.now()+LINK_GRACE_MS;
      try{ render(); }catch(e){}
      linkGraceTimer=setTimeout(async ()=>{
        linkGraceTimer=null;
        linkWaitUntil=0;
        if(pc.connectionState==='connected'){ declareLinkHealthy(); return; }
        const repaired = channelUsable() ? await attemptSilentRepair() : false;
        if(!repaired && pc.connectionState!=='connected') declareLinkLost(pc);
      }, LINK_GRACE_MS);
    }
  };
}

function setupHostChannel(seat, dc, pc){
  monitorConnection(pc, 'graczem');
  NET.seatConn[seat]={dc,pc};
  NET.seatStatus[seat]='pending';
  dc.onmessage=(e)=>{
    let msg; try{ msg=JSON.parse(e.data); }catch(err){ return; }
    if(msg.type==='hello'){
      NET.seatNames[seat]=msg.name || 'Gracz 2';
      NET.seatAvatars=NET.seatAvatars||{};
      NET.seatAvatars[seat]=msg.avatar || AVATARS[1];
      NET.seatStatus[seat]='connected';
      netLinkLost=false;
      if(G && G.mode==='network'){
        /* Reconnecting mid-hand: keep the game and push the current state back. */
        el('netSetup').style.display='none';
        G.message='✓ '+NET.seatNames[seat]+' wrócił do gry.';
        try{ render(); }catch(e){}
      } else {
        showStage('host', 3, '✓ Połączono!', 3);
        el('hostSetupStatus').textContent='Gracie z: '+NET.seatNames[seat]+'. Rozdaję karty…';
        setTimeout(()=>startNetworkGame(), 900);
      }
    } else if(msg.type==='reanswer'){
      /* Friend answered our silent repair offer - apply it, no QR involved. */
      const conn=NET.seatConn[seat];
      if(conn && conn.pc){
        conn.pc.setRemoteDescription({type:'answer', sdp:msg.sdp})
          .then(()=>{ if(G && G.mode==='network'){ G.message='✓ Połączono ponownie — gracz: '+peerName(); try{ render(); }catch(e){} } })
          .catch(()=>{ declareLinkLost(conn.pc); });
      }
    } else if(msg.type==='ping'){
      try{ NET.seatConn[seat].dc.send(JSON.stringify({type:'pong'})); }catch(e){}
      declareLinkHealthy();
    } else if(msg.type==='pong'){
      declareLinkHealthy();
    } else if(msg.type==='action'){
      const resolver=NET.pendingResolvers[seat];
      if(resolver){ NET.pendingResolvers[seat]=null; resolver(msg.action); }
      else { queueRemoteAction(seat, msg.action); }
    }
  };
}

/* Most drops happen when the phone sleeps or the page is backgrounded, so hold a wake lock while
   a networked game is running and re-acquire it whenever the page comes back to the foreground. */
let wakeLock=null;
async function acquireWakeLock(){
  try{
    if('wakeLock' in navigator && !wakeLock){
      wakeLock=await navigator.wakeLock.request('screen');
      wakeLock.addEventListener('release', ()=>{ wakeLock=null; });
    }
  }catch(e){}
}
document.addEventListener('visibilitychange', ()=>{
  if(document.visibilityState==='visible'){
    if(G && G.mode==='network') acquireWakeLock();
    try{ render(); }catch(e){}
  }
});

function startNetworkGame(){
  closeNetSetup();
  acquireWakeLock();
  const friendAvatar=(NET.seatAvatars&&NET.seatAvatars[1])||AVATARS[1];
  if(netDuel){
    G=freshGame('network', 2, [hostChosenName||'Gospodarz', NET.seatNames[1]||'Gracz 2'], 2,
                [hostChosenAvatar, friendAvatar]);
    G.playerControllers=['local','remote'];
  } else {
    G=freshGame('network', 3, [hostChosenName||'Gospodarz', NET.seatNames[1]||'Gracz 2', 'Bot'], 3,
                [hostChosenAvatar, friendAvatar, '🤖']);
    G.playerControllers=['local','remote','bot'];
  }
  runGame();
}

async function clientGenerateAnswerText(text){

function setupClientChannel(dc){
  dc.onopen=()=>{
    dc.send(JSON.stringify({type:'hello', name: clientChosenName, avatar: clientChosenAvatar}));
    try{ showStage('join', 3, '✓ Połączono!', 3); }catch(e){}
    el('netJoinStatus').textContent='Gospodarz rozdaje karty…';
  };
  dc.onmessage=async (e)=>{
    let msg; try{ msg=JSON.parse(e.data); }catch(err){ return; }
    if(msg.type==='state'){
      /* The host rebroadcasts state every couple of seconds as a safety net. That was wiping the
         cards this player had already tapped for the exchange, so the selection could never reach
         two and the hand stalled. Keep local, unsent interaction state across refreshes. */
      const keepExchange = (G && G.exchange && msg.G && msg.G.phase==='exchange'
                            && msg.G.bidder===msg.G.focusSeat && G.exchange.selected
                            && G.exchange.selected.length>0) ? G.exchange : null;
      G=msg.G;
      if(keepExchange) G.exchange=keepExchange;
      pendingAction=null;
      lastStateAt=Date.now();
      stateMsgCount++;
      declareLinkHealthy();
      el('netJoin').style.display='none';
      render();
    } else if(msg.type==='reoffer'){
      /* Host is repairing the link silently - answer over this same channel, no QR. */
      try{
        await NET.pc.setRemoteDescription({type:'offer', sdp:msg.sdp});
        const ans=await NET.pc.createAnswer();
        await NET.pc.setLocalDescription(ans);
        await waitIceComplete(NET.pc);
        dc.send(JSON.stringify({type:'reanswer', sdp:NET.pc.localDescription.sdp}));
        if(G && G.mode==='network'){ G.message='⏳ Odnawiam połączenie — czekam na: '+peerName(); render(); }
      }catch(err){ declareLinkLost(NET.pc); }
    } else if(msg.type==='repair-needed'){
      netLinkLost=true;

function sanitizeStateFor(recipientRealId){
  const clone = JSON.parse(JSON.stringify(G));
  const mySeat = clone.activePlayers.indexOf(recipientRealId);
  for(let s=0;s<(clone.seats||3);s++){
    if(clone.activePlayers[s]!==recipientRealId){
      clone.hands[s]=clone.hands[s].map(()=>({suit:'?',rank:'?'}));
    }

  clone.focusSeat = mySeat;
  /* The face-down musiks must never travel to a player's phone, opened one included until it is
     actually revealed. */

  if(clone.musik && clone.phase!=='musikreveal') clone.musik=clone.musik.map(()=>({suit:'?',rank:'?'}));
  /* Only the bidder (who dealt them) and each recipient may know which card went where. */
  if(clone.receivedCards && mySeat!==clone.bidder){
    const kept={};
    if(clone.receivedCards[mySeat]) kept[mySeat]=clone.receivedCards[mySeat];
    clone.receivedCards=kept;
  }
  return clone;
}

 Nearby Connections łączy telefony bezpośrednio, używając Bluetooth i Wi-Fi Direct.
 Nie potrzeba routera, hotspotu ani internetu, więc działa w samolocie.

 Nazwy metod i zdarzeń są celowo identyczne jak w wersji na iOS, dzięki czemu

@CapacitorPlugin(
    name = "Multipeer",

            alias = "nearby",
            strings = [
                Manifest.permission.ACCESS_FINE_LOCATION,
                Manifest.permission.BLUETOOTH_ADVERTISE,
                Manifest.permission.BLUETOOTH_CONNECT,
                Manifest.permission.BLUETOOTH_SCAN
            ]
        )
    ]
)
class MultipeerPlugin : Plugin() {

    /** Musi być unikalny; Nearby używa go do odsiania obcych aplikacji. */
    private val serviceId = "pl.tysiac.gra"

    /** Gwiazda: jeden gospodarz, reszta dołącza — dokładnie nasz układ. */
    private val strategy = Strategy.P2P_STAR

    private var localName = "Gracz"
    private val discovered = HashMap<String, String>()   // endpointId -> nazwa
    private val connected = HashSet<String>()
    private var pendingCall: PluginCall? = null
    private var pendingMode: String? = null

    private fun client() = Nearby.getConnectionsClient(context)

    // ---------------------------------------------------------------- API

    @PluginMethod
    fun isSupported(call: PluginCall) {
        call.resolve(JSObject().put("supported", true))
    }

    @PluginMethod
    fun startHosting(call: PluginCall) {
        localName = call.getString("displayName") ?: "Gracz"
        pendingCall = call
        pendingMode = "host"
        requestNearbyPermissions(call)
    }

    @PluginMethod
    fun startBrowsing(call: PluginCall) {
        localName = call.getString("displayName") ?: "Gracz"
        pendingCall = call
        pendingMode = "browse"
        requestNearbyPermissions(call)
    }

    @PluginMethod
    fun invite(call: PluginCall) {
        val name = call.getString("peer")
        if (name == null) { call.reject("Brak nazwy gracza"); return }
        val endpoint = discovered.entries.firstOrNull { it.value == name }?.key
        if (endpoint == null) { call.reject("Nie znaleziono tej gry"); return }

        client().requestConnection(localName, endpoint, connectionCallback)
            .addOnSuccessListener { call.resolve(JSObject().put("invited", name)) }
            .addOnFailureListener { e -> call.reject(e.message ?: "Nie udało się połączyć") }
    }

    @PluginMethod
    fun send(call: PluginCall) {
        val text = call.getString("data")
        if (text == null) { call.reject("Brak danych"); return }
        if (connected.isEmpty()) {
            /* Gra i tak powtarza stan co kilka sekund, więc to nie jest błąd krytyczny. */
            call.resolve(JSObject().put("sent", false)); return
        }
        val payload = Payload.fromBytes(text.toByteArray(Charsets.UTF_8))
        client().sendPayload(connected.toList(), payload)
        call.resolve(JSObject().put("sent", true))
    }

    @PluginMethod
    fun stop(call: PluginCall) {
        client().stopAdvertising()
        client().stopDiscovery()
        client().stopAllEndpoints()
        discovered.clear()
        connected.clear()
        call.resolve()
    }

    // -------------------------------------------------------- uprawnienia

    private fun requestNearbyPermissions(call: PluginCall) {
        if (getPermissionState("nearby") != com.getcapacitor.PermissionState.GRANTED) {
            requestPermissionForAlias("nearby", call, "nearbyPermissionCallback")
        } else {
            beginPendingMode()
        }
    }

    @PermissionCallback
    private fun nearbyPermissionCallback(call: PluginCall) {
        if (getPermissionState("nearby") == com.getcapacitor.PermissionState.GRANTED) {
            beginPendingMode()
        } else {
            call.reject("Bez zgody na łączenie z urządzeniami w pobliżu gra nie znajdzie innych graczy")
            pendingCall = null
        }
    }

    private fun beginPendingMode() {
        val call = pendingCall ?: return
        when (pendingMode) {
            "host" -> startAdvertisingNow(call)
            "browse" -> startDiscoveryNow(call)
        }
        pendingCall = null
        pendingMode = null
    }

    // ------------------------------------------------------------ tryby

    private fun startAdvertisingNow(call: PluginCall) {
        val options = AdvertisingOptions.Builder().setStrategy(strategy).build()
        client().startAdvertising(localName, serviceId, connectionCallback, options)
            .addOnSuccessListener {
                call.resolve(JSObject().put("hosting", true).put("name", localName))
            }
            .addOnFailureListener { e ->
                call.reject(e.message ?: "Nie udało się ogłosić gry")
            }
    }

    private fun startDiscoveryNow(call: PluginCall) {
        val options = DiscoveryOptions.Builder().setStrategy(strategy).build()
        client().startDiscovery(serviceId, discoveryCallback, options)
            .addOnSuccessListener { call.resolve(JSObject().put("browsing", true)) }
            .addOnFailureListener { e -> call.reject(e.message ?: "Nie udało się rozpocząć wyszukiwania") }
    }

    // ------------------------------------------------------ nasłuchiwanie

    private val discoveryCallback = object : EndpointDiscoveryCallback() {
        override fun onEndpointFound(endpointId: String, info: DiscoveredEndpointInfo) {
            discovered[endpointId] = info.endpointName
            notifyListeners("peerFound", JSObject().put("name", info.endpointName))
        }

        override fun onEndpointLost(endpointId: String) {
            val name = discovered.remove(endpointId)
            if (name != null) notifyListeners("peerLost", JSObject().put("name", name))
        }
    }

    private val connectionCallback = object : ConnectionLifecycleCallback() {
        override fun onConnectionInitiated(endpointId: String, info: ConnectionInfo) {
            discovered[endpointId] = info.endpointName
            /* Gra jest ogłaszana tylko wtedy, gdy gospodarz sam tego chce, więc przyjmujemy
               połączenie od razu, bez dodatkowego pytania na ekranie. */
            client().acceptConnection(endpointId, payloadCallback)
            notifyListeners("connecting", JSObject().put("name", info.endpointName))
        }

        override fun onConnectionResult(endpointId: String, result: ConnectionResolution) {
            val name = discovered[endpointId] ?: endpointId
            if (result.status.isSuccess) {
                connected.add(endpointId)
                /* Gospodarz przestaje się ogłaszać, gdy komplet już jest przy stole. */
                notifyListeners("connected", JSObject().put("name", name))
            } else {
                notifyListeners("error", JSObject().put("message", "Połączenie odrzucone: $name"))
            }
        }

        override fun onDisconnected(endpointId: String) {
            val name = discovered[endpointId] ?: endpointId
            connected.remove(endpointId)
            notifyListeners("disconnected", JSObject().put("name", name))
        }
    }

    private val payloadCallback = object : PayloadCallback() {
        override fun onPayloadReceived(endpointId: String, payload: Payload) {
            val bytes = payload.asBytes() ?: return
            val text = String(bytes, Charsets.UTF_8)
            val name = discovered[endpointId] ?: endpointId
            notifyListeners("data", JSObject().put("name", name).put("data", text))
        }

        override fun onPayloadTransferUpdate(endpointId: String, update: PayloadTransferUpdate) {
            /* Wiadomości są krótkie i lecą w jednym kawałku, więc postęp nas nie interesuje. */
        }
    }
}

    private var browser: MCNearbyServiceBrowser?

    /// Znalezione urządzenia, żeby JavaScript mógł się odwoływać do nich po nazwie.
    private var discovered: [String: MCPeerID] = [:]

    // MARK: - Cykl życia

    private func makeSession(displayName: String) {
        stopEverything()
        /// Nazwa widoczna dla innych ograniczona do 63 bajtów — tego wymaga system.
        let safeName = String(displayName.prefix(20))
        let pid = MCPeerID(displayName: safeName.isEmpty ? UIDevice.current.name : safeName)
        peerID = pid
        let s = MCSession(peer: pid, securityIdentity: nil, encryptionPreference: .required)
        s.delegate = self
        session = s
    }

    private func stopEverything() {
        advertiser?.stopAdvertisingPeer()
        advertiser?.delegate = nil
        advertiser = nil
        browser?.stopBrowsingForPeers()
        browser?.delegate = nil
        browser = nil
        session?.disconnect()
        session?.delegate = nil
        session = nil
        discovered.removeAll()
    }

    // MARK: - API dla JavaScriptu

    @objc func isSupported(_ call: CAPPluginCall) {
        call.resolve(["supported": true])
    }

    /// Gospodarz: ogłasza grę, żeby inni ją widzieli.
    @objc func startHosting(_ call: CAPPluginCall) {
        let name = call.getString("displayName") ?? "Gracz"
        DispatchQueue.main.async {
            self.makeSession(displayName: name)
            guard let pid = self.peerID else {
                call.reject("Nie udało się utworzyć sesji")
                return
            }
            let adv = MCNearbyServiceAdvertiser(peer: pid,
                                                discoveryInfo: ["gra": "tysiac"],
                                                serviceType: self.serviceType)
            adv.delegate = self
            adv.startAdvertisingPeer()
            self.advertiser = adv
            call.resolve(["hosting": true, "name": pid.displayName])
        }
    }

    /// Dołączający: szuka gier w pobliżu.
    @objc func startBrowsing(_ call: CAPPluginCall) {
        let name = call.getString("displayName") ?? "Gracz"
        DispatchQueue.main.async {
            self.makeSession(displayName: name)
            guard let pid = self.peerID else {
                call.reject("Nie udało się utworzyć sesji")
                return
            }
            let br = MCNearbyServiceBrowser(peer: pid, serviceType: self.serviceType)
            br.delegate = self
            br.startBrowsingForPeers()
            self.browser = br
            call.resolve(["browsing": true])
        }
    }

    /// Dołącza do wybranej gry z listy.
    @objc func invite(_ call: CAPPluginCall) {
        guard let name = call.getString("peer") else {
            call.reject("Brak nazwy gracza")
            return
        }
        DispatchQueue.main.async {
            guard let peer = self.discovered[name],
                  let session = self.session,
                  let browser = self.browser else {
                call.reject("Nie znaleziono tej gry")
                return
            }
            browser.invitePeer(peer, to: session, withContext: nil, timeout: 20)
            call.resolve(["invited": name])
        }
    }

    /// Wysyła wiadomość do wszystkich połączonych urządzeń.
    @objc func send(_ call: CAPPluginCall) {
        guard let text = call.getString("data") else {
            call.reject("Brak danych")
            return
        }
        guard let session = self.session, !session.connectedPeers.isEmpty else {
            /// Nie jest to błąd krytyczny — gra i tak powtarza stan co kilka sekund.
            call.resolve(["sent": false])
            return
        }
        do {
            try session.send(Data(text.utf8), toPeers: session.connectedPeers, with: .reliable)
            call.resolve(["sent": true])
        } catch {
            call.resolve(["sent": false, "error": error.localizedDescription])
        }
    }

    @objc func stop(_ call: CAPPluginCall) {
        DispatchQueue.main.async {
            self.stopEverything()
            call.resolve()
        }
    }
}

/* ============================================================================
   Jeden skrypt konfiguracyjny dla obu systemów.

   `npx cap add` generuje świeże projekty natywne, więc wtyczki, uprawnienia
   i zależności trzeba w nie wpiąć po fakcie. Uruchamiaj z argumentem:

       node scripts/setup.js android
       node scripts/setup.js ios

   Każdy krok jest idempotentny — ponowne uruchomienie niczego nie zdubluje.
   ========================================================================== */

const fs = require('fs');
const path = require('path');
const ROOT = path.join(__dirname, '..');

function fail(msg) { console.error(msg); process.exit(1); }

/* ======================== ANDROID ======================== */

function setupAndroid() {
  
  const PKG_PATH = ['pl', 'tysiac', 'gra'];
  const JAVA_DIR = path.join(ROOT, 'android', 'app', 'src', 'main', 'java', ...PKG_PATH);
  const MANIFEST = path.join(ROOT, 'android', 'app', 'src', 'main', 'AndroidManifest.xml');
  const GRADLE = path.join(ROOT, 'android', 'app', 'build.gradle');
  const SRC_KT = path.join(ROOT, 'native-android', 'MultipeerPlugin.kt');
  
  
  /* --- 1. plik wtyczki --------------------------------------------------- */
  function copyPlugin() {
    if (!fs.existsSync(JAVA_DIR)) fail('Brak projektu Android — uruchom najpierw "npx cap add android".');
    fs.copyFileSync(SRC_KT, path.join(JAVA_DIR, 'MultipeerPlugin.kt'));
    console.log('Skopiowano MultipeerPlugin.kt');
  }
  
  /* --- 2. rejestracja w MainActivity ------------------------------------- */
  function registerPlugin() {
    const main = path.join(JAVA_DIR, 'MainActivity.java');
    const mainKt = path.join(JAVA_DIR, 'MainActivity.kt');
    const target = fs.existsSync(mainKt) ? mainKt : (fs.existsSync(main) ? main : null);
    if (!target) fail('Nie znaleziono MainActivity.');
  
    let src = fs.readFileSync(target, 'utf8');
    if (src.includes('MultipeerPlugin')) {
      console.log('MainActivity już rejestruje wtyczkę — pomijam.');
      return;
    }
  
    if (target.endsWith('.kt')) {
      src = src.replace(
        /class MainActivity[^{]*\{/,
        (m) => m + '\n    override fun onCreate(savedInstanceState: android.os.Bundle?) {\n'
                 + '        registerPlugin(MultipeerPlugin::class.java)\n'
                 + '        super.onCreate(savedInstanceState)\n    }\n'
      );
    } else {
      /* Wariant w Javie: dopisujemy onCreate z rejestracją. */
      src = src.replace(
        /public class MainActivity extends BridgeActivity \{/,
        'public class MainActivity extends BridgeActivity {\n'
        + '    @Override\n'
        + '    public void onCreate(android.os.Bundle savedInstanceState) {\n'
        + '        registerPlugin(MultipeerPlugin.class);\n'
        + '        super.onCreate(savedInstanceState);\n'
        + '    }\n'
      );
    }
    fs.writeFileSync(target, src, 'utf8');
    console.log('Zarejestrowano wtyczkę w MainActivity');
  }
  
  /* --- 3. uprawnienia ---------------------------------------------------- */
  function patchManifest() {
    if (!fs.existsSync(MANIFEST)) fail('Brak AndroidManifest.xml');
    let xml = fs.readFileSync(MANIFEST, 'utf8');
    if (xml.includes('BLUETOOTH_ADVERTISE')) {
      console.log('Manifest ma już uprawnienia — pomijam.');
      return;
    }
  
    /* Bluetooth i Wi-Fi Direct to sposób, w jaki telefony znajdują się bez sieci.
       Na starszych Androidach wymagana jest do tego lokalizacja — taka jest reguła
       systemu, choć gra nie sprawdza, gdzie jesteś. */
    const perms = `
      <uses-permission android:name="android.permission.BLUETOOTH" android:maxSdkVersion="30" />
      <uses-permission android:name="android.permission.BLUETOOTH_ADMIN" android:maxSdkVersion="30" />
      <uses-permission android:name="android.permission.BLUETOOTH_ADVERTISE" />
      <uses-permission android:name="android.permission.BLUETOOTH_CONNECT" />
      <uses-permission android:name="android.permission.BLUETOOTH_SCAN" />
      <uses-permission android:name="android.permission.ACCESS_WIFI_STATE" />
      <uses-permission android:name="android.permission.CHANGE_WIFI_STATE" />
      <uses-permission android:name="android.permission.NEARBY_WIFI_DEVICES" />
      <uses-permission android:name="android.permission.ACCESS_COARSE_LOCATION" android:maxSdkVersion="32" />
      <uses-permission android:name="android.permission.ACCESS_FINE_LOCATION" android:maxSdkVersion="32" />
  `;
    xml = xml.replace('<application', perms + '\n    <application');
    fs.writeFileSync(MANIFEST, xml, 'utf8');
    console.log('Dodano uprawnienia do AndroidManifest');
  }
  
  /* --- 4. biblioteka Nearby --------------------------------------------- */
  function patchGradle() {
    if (!fs.existsSync(GRADLE)) fail('Brak build.gradle');
    let g = fs.readFileSync(GRADLE, 'utf8');
    if (g.includes('play-services-nearby')) {
      console.log('build.gradle ma już bibliotekę Nearby — pomijam.');
      return;
    }
    g = g.replace(
      /dependencies\s*\{/,
      "dependencies {\n    implementation 'com.google.android.gms:play-services-nearby:19.3.0'"
    );
    fs.writeFileSync(GRADLE, g, 'utf8');
    console.log('Dodano bibliotekę Nearby do build.gradle');
  }
  
  
  copyPlugin();
  registerPlugin();
  patchManifest();
  patchGradle();
  console.log('Projekt Android gotowy.');
}

/* ========================== iOS ========================== */

function setupIos() {
  const SRC = path.join(ROOT, 'native', 'MultipeerPlugin.swift');
  const DEST_DIR = path.join(ROOT, 'ios', 'App', 'App');
  if (!fs.existsSync(DEST_DIR)) fail('Brak projektu iOS — uruchom najpierw "npx cap add ios".');
  fs.copyFileSync(SRC, path.join(DEST_DIR, 'MultipeerPlugin.swift'));
  console.log('Skopiowano MultipeerPlugin.swift');

  const PLIST = path.join(DEST_DIR, 'Info.plist');
  if (!fs.existsSync(PLIST)) fail('Brak Info.plist');
  let xml = fs.readFileSync(PLIST, 'utf8');
  if (xml.includes('NSLocalNetworkUsageDescription')) {
    console.log('Info.plist ma już uprawnienia — pomijam.');
  } else {
    const additions = [
      '\t<key>NSLocalNetworkUsageDescription</key>',
      '\t<string>Aplikacja wyszukuje gry innych graczy w pobliżu, żeby móc grać bez internetu. Dane nie opuszczają Waszych telefonów.</string>',
      '\t<key>NSBonjourServices</key>',
      '\t<array>',
      '\t\t<string>_tysiac._tcp</string>',
      '\t\t<string>_tysiac-gra._tcp</string>',
      '\t\t<string>_tysiac-gra._udp</string>',
      '\t</array>',
      '\t<key>UIRequiresFullScreen</key>',
      '\t<true/>',
      ''
    ].join('\n');
    const marker = '</dict>\n</plist>';
    const idx = xml.lastIndexOf(marker);
    if (idx === -1) fail('Nieoczekiwany format Info.plist — przerywam, żeby go nie uszkodzić.');
    xml = xml.slice(0, idx) + additions + xml.slice(idx);
    fs.writeFileSync(PLIST, xml, 'utf8');
    console.log('Dodano uprawnienia sieci lokalnej do Info.plist');
  }
  console.log('Projekt iOS gotowy.');
}

const target = process.argv[2];
if (target === 'android') setupAndroid();
else if (target === 'ios') setupIos();
else fail('Podaj system: node scripts/setup.js android|ios');

# Tysiąc — wersja na App Store

Projekt gotowy do zbudowania **bez komputera**. Wszystko robisz w przeglądarce na telefonie.

---

## Co jest w środku

| Plik | Do czego służy |
|---|---|
| `www/` | Cała gra — działa w 100% offline, nic nie pobiera z sieci |
| `www/nearby.js` | **Łączenie bez sieci** — Bluetooth i Wi-Fi peer-to-peer (tryb do samolotu) |
| `native/MultipeerPlugin.swift` | Wtyczka łączenia bez sieci — iOS |
| `native-android/MultipeerPlugin.kt` | Wtyczka łączenia bez sieci — Android |
| `www/native-net.js` | Zapasowo: wyszukiwanie przez Wi-Fi (Bonjour + WebSocket) |

| `scripts/patch-ios-plist.js` | Dopisuje wymagane uprawnienia sieci lokalnej |
| `codemagic.yaml` | Budowanie i wysyłka do App Store w chmurze |

---

## Zacznij od Androida

Android jest **znacznie szybszy do sprawdzenia** i nic nie kosztuje:

- nie potrzeba konta w sklepie ani recenzji,
- dostajesz plik **APK** i instalujesz go bezpośrednio na telefonie,
- ta sama gra i ten sam mechanizm łączenia bez sieci, co na iOS.

**Jak to zrobić:**
1. Wgraj projekt na GitHub (patrz niżej).
2. W Codemagic uruchom workflow **`android-apk`** — wcześniej podmień adres e-mail
   w `codemagic.yaml` na swój.
3. Po ~10 minutach dostaniesz mailem link do pliku APK.
4. Otwórz link na telefonie z Androidem i zainstaluj (system poprosi o zgodę
   na instalację z nieznanego źródła — trzeba się zgodzić).
5. Przy pierwszym uruchomieniu zgódź się na **Bluetooth i urządzenia w pobliżu**.

**Uwaga o testowaniu:** żeby sprawdzić granie we dwoje bez sieci, potrzebne są
**dwa telefony z Androidem**. Mając jeden, zobaczysz działającą grę i tryby solo,
ale połączenia między telefonami nie przetestujesz. Aplikacja na Androida
**nie połączy się** z aplikacją na iPhonie — to dwa różne systemy łączenia.

## Czego brakuje po Twojej stronie

**Tylko dla iOS — konto Apple Developer, 99 USD/rok.** Do wersji na Androida nie jest potrzebne.
Bez konta Apple nie da się nic opublikować w App Store. Rejestracja
przez aplikację **Apple Developer** na iPhonie. Weryfikacja trwa zwykle 1–2 dni.

---

## Krok po kroku

### 1. GitHub (przeglądarka na telefonie)
1. Załóż konto na `github.com`.
2. Utwórz **prywatne** repozytorium `tysiac`.
3. Wgraj zawartość tej paczki (przycisk *Add file → Upload files*).

### 2. Klucz App Store Connect
1. `appstoreconnect.apple.com` → **Users and Access** → **Integrations** → **App Store Connect API**.
2. Utwórz klucz z rolą **App Manager**.
3. Zapisz **Issuer ID**, **Key ID** i pobrany plik **.p8** (pobierzesz go tylko raz).

### 3. Rejestracja aplikacji
1. W App Store Connect → **Apps** → **+** → **New App**.
2. Bundle ID: `pl.tysiac.gra` (utwórz go wcześniej w portalu Certificates, Identifiers & Profiles).
3. Nazwa: `Tysiąc`, język: polski, kategoria: **Games → Card**.

### 4. Codemagic
1. `codemagic.io` → zaloguj przez GitHub → dodaj repozytorium.
2. **Teams → Integrations → App Store Connect** → wklej Issuer ID, Key ID i plik .p8.
   Nazwij integrację dokładnie **`TysiacKey`** (tak jest w `codemagic.yaml`).
3. Utwórz grupę zmiennych **`appstore`**, a w niej `APP_STORE_APP_ID` — numer aplikacji
   widoczny w adresie App Store Connect.
4. Uruchom workflow **ios-appstore**.

Pierwsza kompilacja trwa 10–20 minut. Efekt trafia do **TestFlight**, gdzie zainstalujecie
grę na swoich telefonach i sprawdzicie ją na żywo.

### 5. Wysyłka do recenzji
Gdy TestFlight potwierdzi, że wszystko działa:
1. W `codemagic.yaml` zmień `submit_to_app_store` na `true`.
2. W App Store Connect uzupełnij opis, zrzuty ekranu (zrobisz je telefonem),
   politykę prywatności i **Age Rating**.
3. Wyślij do recenzji. Apple odpowiada zwykle w 1–3 dni.

---

## Dwa sposoby łączenia — który wybrać

W grze są dwa tryby, przełączane w lobby. Żaden nie wymaga internetu.

### 📶 W pobliżu — bez żadnej sieci
Telefony łączą się bezpośrednio przez Bluetooth i Wi-Fi peer-to-peer, jak przy AirDropie.
Nie trzeba niczego konfigurować.

**Ograniczenie: tylko ten sam system.** iPhone połączy się z iPhonem, Android z Androidem.
Apple i Google używają tu zamkniętych, niezgodnych ze sobą mechanizmów (MultipeerConnectivity
kontra Nearby Connections) i nie da się ich pogodzić.

### 🔗 iPhone ↔ Android — przez hotspot
Jeden telefon włącza **hotspot osobisty**, drugi łączy się z nim przez Wi-Fi. Powstaje sieć
lokalna — **pakiet danych ani internet nie są potrzebne**, sam hotspot wystarczy. Gra wykrywa
przeciwnika przez Bonjour i łączy się przez WebSocket, a oba te standardy działają tak samo
na iOS i na Androidzie.

To **jedyny sposób na grę między iPhonem a Androidem** i działa w samolocie.

> Uwaga: w przeglądarce ten tryb zawodził, bo Safari ukrywa adres telefonu nadającego hotspot.
> W aplikacji natywnej to ograniczenie nie występuje — mamy pełny dostęp do gniazd sieciowych.

## Jak działa łączenie bez sieci

W samolocie nie ma Wi-Fi ani internetu, więc gra używa **MultipeerConnectivity** —
tego samego mechanizmu co AirDrop. Telefony łączą się bezpośrednio przez Bluetooth
i Wi-Fi peer-to-peer. Nie potrzeba routera, hotspotu ani jakiejkolwiek sieci.

W praktyce: jeden zakłada grę, drugi widzi ją na liście „Gry w pobliżu" i dołącza.
Zasięg to kilkanaście metrów, więc przy sąsiednich fotelach jest z zapasem.

Przy pierwszym uruchomieniu iOS zapyta o zgodę na łączenie z urządzeniami w pobliżu —
**trzeba się zgodzić**, inaczej lista pozostanie pusta.

Gdyby na jakimś urządzeniu ten tryb nie działał, gra automatycznie użyje wyszukiwania
przez wspólne Wi-Fi, a w ostateczności parowania kodem.

## Uczciwie o ryzyku

**Apple potrafi odrzucać aplikacje będące opakowaną stroną internetową** (zasada 4.2,
*Minimum Functionality*). Nasza gra jest w znacznie lepszej sytuacji niż typowy taki
przypadek, bo:

- działa **całkowicie offline**, nie wczytuje żadnego adresu z internetu,
- ma **funkcję niedostępną dla stron** — wyszukiwanie gier w sieci lokalnej,
- to pełna gra z własną logiką, a nie skrót do serwisu.

Mimo to gwarancji nie ma. Jeśli recenzent odrzuci wersję, dostaniesz konkretny powód
i wtedy dobudujemy to, czego zażąda.

**Druga sprawa:** wtyczki natywnej nie dało się skompilować w tym środowisku — nie ma tu
Xcode. Kod Swift jest napisany według standardowego API MultipeerConnectivity, ale
**pierwsza kompilacja w Codemagic jest jego pierwszym prawdziwym testem**. Gdyby coś
nie zagrało, błąd pojawi się w logu kompilacji i wtedy to poprawimy — gra ma dwie
ścieżki zapasowe, więc nie zostaniesz bez działającej wersji.

---

## Kolejność, którą polecam

1. TestFlight na dwa telefony → sprawdźcie wyszukiwanie gier w sieci.
2. Dopiero potem recenzja App Store.

Dzięki temu ewentualne problemy z siecią lokalną wyjdą, zanim aplikację zobaczy recenzent.
