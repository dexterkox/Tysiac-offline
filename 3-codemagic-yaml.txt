# =============================================================================
# Budowanie i wysyłka do App Store bez Maca.
#
# Codemagic uruchamia prawdziwy komputer Mac w chmurze, generuje projekt iOS,
# podpisuje go Twoim certyfikatem i wysyła do App Store Connect. Ty klikasz
# wyłącznie w przeglądarce.
#
# Zanim to zadziała, w panelu Codemagic trzeba raz ustawić:
#   1. Integrację App Store Connect (klucz API: Issuer ID, Key ID, plik .p8)
#   2. Zmienną BUNDLE_ID = pl.tysiac.gra
#   3. Grupę zmiennych o nazwie "appstore"
# =============================================================================

workflows:
  ios-appstore:
    name: Tysiąc — iOS App Store
    max_build_duration: 60
    instance_type: mac_mini_m2

    integrations:
      app_store_connect: TysiacKey   # nazwa integracji utworzonej w Codemagic

    environment:
      groups:
        - appstore
      vars:
        BUNDLE_ID: "pl.tysiac.gra"
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
      node: 20
      xcode: latest
      cocoapods: default

    scripts:
      - name: Instalacja zależności
        script: |
          npm install

      - name: Wygenerowanie projektu iOS
        script: |
          npx cap add ios || echo "Projekt iOS już istnieje"
          npx cap sync ios

      - name: Wtyczka łączenia bez sieci
        script: |
          node scripts/setup.js ios
          # Dodaje plik Swift do celu kompilacji (cap add ios go nie zna).
          cd ios/App
          ruby -e "
            require 'xcodeproj'
            project = Xcodeproj::Project.open('App.xcodeproj')
            target  = project.targets.find { |t| t.name == 'App' }
            group   = project.main_group.find_subpath('App', true)
            unless target.source_build_phase.files_references.any? { |f| f.path.to_s.include?('MultipeerPlugin') }
              ref = group.new_file('MultipeerPlugin.swift')
              target.add_file_references([ref])
              project.save
              puts 'Dodano MultipeerPlugin.swift do celu kompilacji.'
            else
              puts 'MultipeerPlugin.swift już jest w projekcie.'
            end
          "

      - name: Numer kompilacji
        script: |
          # Każda wysyłka do App Store musi mieć wyższy numer niż poprzednia.
          BUILD_NUMBER=$(($(app-store-connect get-latest-app-store-build-number "$APP_STORE_APP_ID" 2>/dev/null || echo 0) + 1))
          echo "Numer kompilacji: $BUILD_NUMBER"
          cd ios/App
          agvtool new-version -all "$BUILD_NUMBER"

      - name: Podpisywanie
        script: |
          keychain initialize
          app-store-connect fetch-signing-files "$BUNDLE_ID" \
            --type IOS_APP_STORE \
            --create
          keychain add-certificates
          xcode-project use-profiles

      - name: Instalacja podów
        script: |
          cd ios/App && pod install

      - name: Budowanie
        script: |
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"

    artifacts:
      - build/ios/ipa/*.ipa
      - /tmp/xcodebuild_logs/*.log

    publishing:
      app_store_connect:
        auth: integration
        # Trafia do TestFlight, żeby najpierw sprawdzić grę na własnych telefonach.
        submit_to_testflight: true
        # Dopiero po testach przełącz na true, żeby iść do recenzji App Store.
        submit_to_app_store: false

  # ===========================================================================
  # Android — najszybsza droga do sprawdzenia gry na żywo.
  # Nie wymaga konta w sklepie ani recenzji: dostajesz plik APK i instalujesz go
  # bezpośrednio na telefonie.
  # ===========================================================================
  android-apk:
    name: Tysiąc — Android APK (do testów)
    max_build_duration: 45
    instance_type: linux_x2

    environment:
      node: 20
      java: 17

    scripts:
      - name: Instalacja zależności
        script: |
          npm install

      - name: Wygenerowanie projektu Android
        script: |
          npx cap add android || echo "Projekt Android już istnieje"
          npx cap sync android

      - name: Wtyczka łączenia bez sieci
        script: |
          node scripts/setup.js android

      - name: Budowanie APK
        script: |
          cd android
          chmod +x ./gradlew
          # Wersja debug jest podpisana kluczem testowym, więc instaluje się od razu.
          ./gradlew assembleDebug

    artifacts:
      - android/app/build/outputs/**/*.apk

    publishing:
      email:
        recipients:
          - CHANGE_ME@example.com     # tu przyjdzie link do gotowego APK
        notify:
          success: true
          failure: true
